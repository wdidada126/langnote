根据加锁的范围，MySQL 里面的锁大致可以分成全局锁、表级锁和行锁三类

CREATE TABLE `T_ETP_INNER_ACC_BOOK_D` (
  `BOOK_ID` varchar(36) COLLATE utf8mb4_bin NOT NULL COMMENT '记账ID',
  `ETP_ID` varchar(36) COLLATE utf8mb4_bin NOT NULL COMMENT '企业ID',
  `ACC_ID` varchar(36) COLLATE utf8mb4_bin NOT NULL COMMENT '账户ID',
  `OPST_ACC_NO` varchar(36) COLLATE utf8mb4_bin DEFAULT NULL COMMENT '对方账号',
  `OPST_ACC_NAME` varchar(256) COLLATE utf8mb4_bin DEFAULT NULL COMMENT '对方户名',
  `TRANS_TIME` varchar(19) COLLATE utf8mb4_bin NOT NULL COMMENT '交易时间',
  `TRANS_TYPE` varchar(16) COLLATE utf8mb4_bin NOT NULL COMMENT '交易类型：收入INCOME、支出PAY',
  `TRANS_AMOUNT` decimal(14,2) NOT NULL COMMENT '交易金额',
  `ACTUAL_TRANS_AMOUNT` decimal(14,2) DEFAULT NULL COMMENT '实际交易金额',
  `TRANS_BALANCE` decimal(14,2) NOT NULL COMMENT '交易余额',
  `TRANS_VOUC_NAME` varchar(256) COLLATE utf8mb4_bin DEFAULT NULL COMMENT '交易凭证名称',
  `TRANS_VOUC_URL` varchar(256) COLLATE utf8mb4_bin DEFAULT NULL COMMENT '交易凭证路径',
  `TRANS_STATUS` varchar(16) COLLATE utf8mb4_bin DEFAULT NULL COMMENT '交易状态：成功SUCCESSED、失败FAILED、待支付WAITPAY、部分成功PARTSUCCESSED',
  `REMARK` varchar(1024) COLLATE utf8mb4_bin DEFAULT NULL COMMENT '备注',
  `COMMISSION_ID` varchar(36) COLLATE utf8mb4_bin DEFAULT NULL COMMENT '代发ID',
  `CREATE_MAN` varchar(36) COLLATE utf8mb4_bin NOT NULL COMMENT '创建人',
  `CREATE_TIME` varchar(19) COLLATE utf8mb4_bin NOT NULL COMMENT '创建时间',
  `UPDATE_MAN` varchar(36) COLLATE utf8mb4_bin NOT NULL COMMENT '更新人',
  `UPDATE_TIME` varchar(19) COLLATE utf8mb4_bin NOT NULL COMMENT '更新时间',
  `TRAD_FLOW_NO` varchar(36) COLLATE utf8mb4_bin DEFAULT NULL COMMENT '交易流水号',
  `TRADE_AIM` varchar(128) COLLATE utf8mb4_bin DEFAULT NULL COMMENT '交易用途',
  `WRITEOFF_STAT` varchar(1) COLLATE utf8mb4_bin DEFAULT '0' COMMENT '核销状态 0未核销 1部分核销 2 全部核销',
  `CREATE_TYPE` varchar(1) COLLATE utf8mb4_bin DEFAULT NULL COMMENT '生成方式：空：自主生成 1：导入生成',
  `TO_WRIOFF_AMT` decimal(16,2) DEFAULT NULL COMMENT '待核销金额',
  `WRIOFF_AMT` decimal(16,2) DEFAULT '0.00' COMMENT '已核销金额',
  `OPP_OPEN_BANK` varchar(256) COLLATE utf8mb4_bin DEFAULT NULL COMMENT '对方开户行',
  PRIMARY KEY (`BOOK_ID`),
  KEY `INDEX_ETP_ID` (`ETP_ID`),
  KEY `INDEX_ACC_ID` (`ACC_ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_bin COMMENT='企业内部账户记账表';
