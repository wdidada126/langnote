Redis is an open-source, in-memory data structure store, used as a database, cache, and message broker. The Redis source code is available on GitHub, and you can download it from the following link: https://github.com/redis/redis.
Redis includes a unit test suite that covers various parts of the Redis codebase, including the quicklist data structure. The quicklist is a data structure used to store a list of items in Redis, and it's optimized for small lists.
The quicklist unit test code can be found in the tests/unit/quicklist.c file in the Redis source code. The quicklist test suite includes tests for creating and deleting quicklists, adding and removing items, iterating over the list, and more.

static void test_zset_add_get_rank(void) {
    zset *zset = zsetCreate();
    double score;
    robj *ele;
    int rank;

    ele = createStringObject("one", 3);
    score = 1.0;
    zsetAdd(zset, score, ele);
    decrRefCount(ele);
    ele = createStringObject("two", 3);
    score = 2.0;
    zsetAdd(zset, score, ele);
    decrRefCount(ele);
    ele = createStringObject("three", 5);
    score = 3.0;
    zsetAdd(zset, score, ele);
    decrRefCount(ele);

    ele = createStringObject("one", 3);
    rank = zsetRank(zset, ele);
    assert(rank == 0);
    decrRefCount(ele);
    ele = createStringObject("two", 3);
    rank = zsetRank(zset, ele);
    assert(rank == 1);
    decrRefCount(ele);
    ele = createStringObject("three", 5);
    rank = zsetRank(zset, ele);
    assert(rank == 2);
    decrRefCount(ele);

    zsetFree(zset);
}

Redis includes a unit test suite that covers various parts of the Redis codebase, including the sorted set data structure. static void test_quicklist_pushpop(void) {
    quicklist *quicklist = quicklistNew(4, 2);
    unsigned char *value;
    int i;

    for (i = 0; i < 10; i++) {
        value = (unsigned char *) zmalloc(sizeof(unsigned char));
        *value = i;
        quicklistPush(quicklist, value, sizeof(unsigned char));
    }

    for (i = 9; i >= 0; i--) {
        quicklistPop(quicklist, &value, NULL);
        assert(*value == i);
        zfree(value);
    }

    assert(quicklist->len == 0);
    assert(quicklist->count == 1);
    quicklistRelease(quicklist);
}
