import javax.validation.ConstraintViolation;
import javax.validation.Validation;
import javax.validation.Validator;
import javax.validation.ValidatorFactory;
import java.util.Set;

public class ValidatorDemo {

    public static void main(String[] args) {
        // 创建一个 ValidatorFactory 对象
        ValidatorFactory factory = Validation.buildDefaultValidatorFactory();
        // 从 ValidatorFactory 中获取一个 Validator 对象
        Validator validator = factory.getValidator();
        // 创建一个 Car 对象，用于校验
        Car car = new Car(null, "ABC123", 4);
        // 使用 Validator 的 validate 方法来校验 Car 对象是否符合约束条件
        Set<ConstraintViolation<Car>> violations = validator.validate(car);
        // 如果 violations 集合不为空，表示有校验错误
        if (!violations.isEmpty()) {
            // 遍历 violations 集合，打印出每个校验错误的信息
            for (ConstraintViolation<Car> violation : violations) {
                System.out.println(violation.getPropertyPath() + ": " + violation.getMessage());
            }
            // 抛出一个自定义的异常，表示校验失败
            throw new ValidationException("Car is not valid");
        }
    }
}
