SELECT DISTINCT
	hdr.from_bill AS FROM_BILL,
	hdr.co_num_id AS CO_NUM_ID,
	hdr.type_num_id AS TYPE_NUM_ID,
	hdr.cort_num_id AS CORT_NUM_ID 
FROM
	sd_bl_co_hdr hdr
	LEFT JOIN sd_bl_co_dtl dtl ON hdr.co_num_id = dtl.co_num_id 
WHERE
	hdr.cancelsign = 'N' 
	AND dtl.cancelsign = 'N' 
	AND hdr.from_bill is not null
	AND hdr.from_bill != ""
	AND hdr.cort_num_id = 'HT03'
