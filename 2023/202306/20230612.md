BatchPreparedStatementSetter


```java
    public Integer batAudit(List<Long> seriesList) {
        String sql = "UPDATE " + getTableName() + " SET STAT = ? WHERE STAT="+Constant.NOT_AUDIT+" and cancelsign ='N' and series IN (" + String.join(",", Collections.nCopies(seriesList.size(), "?")) + ")";
        int[] resultSize = jdbcTemplate.batchUpdate(sql, new BatchPreparedStatementSetter() {
            @Override
            public void setValues(PreparedStatement ps, int i) throws SQLException {
                ps.setInt(1, Constant.AUDIT);
                for(int index = 0;index<seriesList.size();index++){
                    ps.setLong(2+index, seriesList.get(index));
                }
            }
            @Override
            public int getBatchSize() {
                return seriesList.size();
            }
        });
        return resultSize[0];
    }
```
