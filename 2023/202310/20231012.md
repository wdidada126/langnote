import java.util.List;  
import java.util.stream.Collectors;  
  
public class Main {  
    public static void main(String[] args) {  
        List<FI_B2C_INVOICE> invoiceWeChatWarnList = ...; // 获取FI_B2C_INVOICE列表  

        List<FI_B2C_INVOICE> nonNullInvoices = invoiceWeChatWarnList.stream() // 使用流API  
                                                       .filter(Objects::nonNull) // 过滤非null元素  
                                                       .collect(Collectors.toList()); // 收集非null元素到新列表

        // 如果需要获取前10个非null元素，可以继续对nonNullInvoices进行切片操作
        List<FI_B2C_INVOICE> top10NonNullInvoices = nonNullInvoices.stream()
                                                       .limit(10) // 取前10个元素
                                                       .collect(Collectors.toList());
    }
}
