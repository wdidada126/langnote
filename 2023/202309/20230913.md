    /*
     * 2023-10返回2023-09
     */
    public static String getPreviousMonth2(String date) {
        if(date==null || date.trim().length() == 0){
            return null;
        }
        DateTimeFormatter formatter = DateTimeFormatter.ofPattern("yyyy-MM-dd");
        LocalDate localDate = LocalDate.parse(date + "-01", formatter);
        localDate = localDate.minusMonths(1);
        return localDate.format(formatter).substring(0,7);//2023-09 -> 2023-09-1
    }

    /*
    * 2023-10返回2023-09
     */
    public static String getPreviousMonth(String date) {
        if (date == null || date.trim().length() == 0) {
            return null;
        }

        SimpleDateFormat formatter = new SimpleDateFormat(DATE_FORMAT_PATTERN);
        Date inputDate;
        try {
            inputDate = formatter.parse(date);
        } catch (ParseException e) {
            e.printStackTrace();
            return null;
        }

        Calendar calendar = Calendar.getInstance();
        calendar.setTime(inputDate);

        calendar.add(Calendar.MONTH, -1);

        return formatter.format(calendar.getTime());
    }
