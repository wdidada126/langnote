  快速排序的基本原理这里就不详细介绍了，下面给出快排的几种实现方式，以及优化。

 
### 单项Partition

 对数组分段的时候我们采用从一端开始的方式，下面给出代码

 
```
private int partion1Way(int[]a,int start,int end){
        int i = start-1;
        int j = start;
        int k = a[start];
        for(;j<=end;j++){
            if(a[j]<=k){
                swap(a,j,++i);
            }
        }
        swap(a,start,i);
        //最后形成的数组[start,i-1] <= [i] <=[i+1,end]
        return i;
    }
```
 
### 双向partition

 
```
private int partion2Way(int[]a,int start,int end){
        int key = a[start];
        int i = start;
        int j = end+1;
        while (i < j) {
            while(a[++i] <= key)
                if(i >= end) break;
            while(a[--j]>=key)
                if(j<=start) break;
            if(i < j)
                swap(a,i,j);
        }
        swap(a,start,j);
        //最后形成的数组[start,j-1] <= [j] <=[j+1,end]
        return j;
    }
```
 
### 完整代码

 
```
public void sort(int []a){
        sort(a,0,a.length-1);
 }

    private void sort(int[]a,int start,int end){
        if(start >= end) return;
        //这里可以条用任意一种partition方法
        //int mid = partion2Way(a,start,end);
        int mid = partion1Way(a,start,end);
        sort(a,start,mid-1);
        sort(a,mid+1,end);
    }
```
 
### 三项快速排序

 当数组中存在大量相同元素的时候，我们可以将数组分成三段，前段比Key值小，中间段等于Key,后段比Key大。代码如下：

 
```
private void sort3Way(int[]a,int start,int end) {
        if(start>=end) return;
        int key = a[start];
        int i = start;
        int j = start+1;
        int gt = end;
        while(j<=gt){
            if(a[j]<key){
                swap(a,i++,j++);
            }else if(a[j]>key){
                swap(a,j,gt--);
            }else{
                j++;
            }
        }
        sort3Way(a,start,i-1);
        sort3Way(a,gt+1,end);
    }
public void sort(int []a){
        sort3Way(a,0,a.length-1);
    }
```
 
### 快排的优化

 当数组元素小于等于15元素的时候，插入排序的算法是很好的，甚至优于快速排序，所以我们当分段中数组小于等于15的时候，可以直接使用插入排序。

 
```
private void sortWithInsert(int[]a,int start,int end){
        int m = 15;
        if(start <= end + m) {
            insertSort(a,start,end);
        }else{
            int mid = partion2Way(a,start,end);
            sortWithInsert(a,start,mid-1);
            sortWithInsert(a,mid+1,end);
        }
    }
private void insertSort(int[]a,int start,int end){
        for(int i = start+1;i<=end;i++){
            int key = a[i];
            int j;
            for(j=i-1;j>=start;j--){
                if(key<a[j]){
                    a[j+1] = a[j];
                }else{
                    break;
                }
            }
            a[j+1] = key;
        }
    }
```
