# 第 28 章 项目9：使用GUI共享文件（原书 pp.417-422）

> 一句话定位本章：给第 27 章的 XML-RPC 节点套上一个 **Tkinter 桌面界面**，让「启动节点 / 看文件列表 / 下载别人的文件」变成点按钮——全书最短的一章，但示范了「网络程序 + 图形界面」如何共存。

## 核心精讲

> 以下代码均为**教学示意，不参与构建**：示例用于对齐概念与 API 形态，不发布为真实软件。

### 28.2–28.3 界面背后的三条硬约束

**GUI 必须在主线程**（Tk 事件循环非线程安全）→ 所有控件操作都在主线程完成；
**网络/磁盘不能堵住 UI**（阻塞会让窗口「未响应」）→ 耗时操作丢进 `threading.Thread(daemon=True)`；
**子线程不能直接碰控件**（跨线程调用 Tk 会随机崩溃）→ 用 `root.after(0, callback)` 把结果交回主线程。

### 28.4 初次实现：最小窗口（列表 + 下载）

```python
# node_gui_min.py —— 教学示意，不参与构建
# 纯标准库（tkinter）。Linux 若缺 Tk：`sudo apt install python3-tk`
from __future__ import annotations

import tkinter as tk
from pathlib import Path
from tkinter import messagebox
from xmlrpc.client import Fault, ProtocolError, ServerProxy

PEER = "http://127.0.0.1:8000/RPC2"
OUTDIR = Path("downloads")

def list_local() -> list[str]:
    shared = Path("shared"); shared.mkdir(exist_ok=True)
    return sorted(str(p.relative_to(shared)) for p in shared.rglob("*") if p.is_file())

class App:
    def __init__(self, root: tk.Tk) -> None:
        root.title("文件共享节点（最小版）")
        self.box = tk.Listbox(root, width=60, height=12)
        self.box.pack(padx=8, pady=8)
        for name in list_local():
            self.box.insert(tk.END, name)
        tk.Button(root, text="下载选中文件", command=self.on_download).pack(pady=4)

    def on_download(self) -> None:
        if not (selection := self.box.curselection()):
            return messagebox.showinfo("提示", "请先选中一个文件")
        name = self.box.get(selection[0])
        try:
            data = ServerProxy(PEER, allow_none=True).fetch(name, "change-me")
            OUTDIR.mkdir(exist_ok=True)
            dest = (OUTDIR / Path(name).name).resolve()
            if not dest.is_relative_to(OUTDIR.resolve()):      # 落盘也要防穿越
                raise ValueError("拒绝写入：目标路径越界")
            dest.write_bytes(data.data if hasattr(data, "data") else data)
        except Fault as exc:
            messagebox.showerror("服务端错误", f"[{exc.faultCode}] {exc.faultString}")
        except ProtocolError as exc:
            messagebox.showerror("连接错误", exc.errmsg)
        except Exception as exc:                               # 兜底：界面永不因异常崩掉
            messagebox.showerror("失败", str(exc))
        else:
            messagebox.showinfo("完成", f"已保存到 {dest}")

if __name__ == "__main__":
    root = tk.Tk(); App(root); root.mainloop()
```

### 28.5 再次实现：服务器开关 + URL 输入 + 子线程 + 状态栏

```python
# node_gui.py —— 教学示意，不参与构建
# 运行：python node_gui.py
from __future__ import annotations

import os
import threading
from pathlib import Path
from xmlrpc.client import Binary, Fault, ProtocolError, ServerProxy
from xmlrpc.server import SimpleXMLRPCRequestHandler, SimpleXMLRPCServer
import tkinter as tk
from tkinter import messagebox, ttk

SHARED = Path("shared").resolve()
OUTDIR = Path("downloads").resolve()
SECRET = os.environ.get("NODE_SECRET", "change-me")

class Node:                                        # 与第 27 章同源的节点逻辑
    def list(self) -> list[str]:
        return sorted(str(p.relative_to(SHARED)) for p in SHARED.rglob("*") if p.is_file())

    def fetch(self, query: str, secret: str) -> Binary:
        if secret != SECRET:
            raise Fault(2, "密钥不匹配")
        path = (SHARED / query).resolve()
        if not path.is_relative_to(SHARED):        # 路径穿越校验，详见第 27 章
            raise Fault(1, f"非法路径：{query!r}")
        if not path.is_file():
            raise Fault(3, f"文件不存在：{query}")
        return Binary(path.read_bytes())

class Handler(SimpleXMLRPCRequestHandler):
    rpc_paths = ("/RPC2",)

class App:
    def __init__(self, root: tk.Tk) -> None:
        self.root, self.server = root, None
        root.title("文件共享节点")
        form = ttk.Frame(root, padding=6); form.pack(fill=tk.X)
        ttk.Label(form, text="对端 URL").grid(row=0, column=0)
        self.peer = ttk.Entry(form, width=42)
        self.peer.insert(0, "http://127.0.0.1:8000/RPC2"); self.peer.grid(row=0, column=1)
        bar = ttk.Frame(root, padding=6); bar.pack(fill=tk.X)
        self.btn_serve = ttk.Button(bar, text="启动本节点", command=self.toggle_server)
        self.btn_serve.pack(side=tk.LEFT)
        ttk.Button(bar, text="刷新列表", command=self.refresh).pack(side=tk.LEFT)
        ttk.Button(bar, text="下载选中", command=self.download).pack(side=tk.LEFT)
        self.box = tk.Listbox(root, width=64, height=14)
        self.box.pack(padx=6, pady=4, fill=tk.BOTH, expand=True)
        self.status = tk.StringVar(value="就绪")
        ttk.Label(root, textvariable=self.status, anchor=tk.W).pack(fill=tk.X, padx=6)
        SHARED.mkdir(exist_ok=True)
        self.refresh()

    def _post(self, fn) -> None:
        """子线程 → 主线程：所有控件操作都必须经过这里（Tk 非线程安全）。"""
        self.root.after(0, fn)

    def toggle_server(self) -> None:
        if self.server is not None:                            # —— 停止 ——
            srv, self.server = self.server, None
            threading.Thread(target=srv.shutdown, daemon=True).start()
            self.btn_serve.configure(text="启动本节点")
            return self.status.set("本节点已停止")

        def serve() -> None:                                   # —— 启动（跑在子线程）——
            srv = SimpleXMLRPCServer(("127.0.0.1", 8000), Handler,
                                     allow_none=True, logRequests=False)
            srv.register_instance(Node()); self.server = srv
            self._post(lambda: self.status.set("本节点已启动：8000"))
            srv.serve_forever()

        threading.Thread(target=serve, daemon=True).start()    # daemon：关窗即退
        self.btn_serve.configure(text="停止本节点")

    def refresh(self) -> None:
        def work() -> None:
            try:
                names = ServerProxy(self.peer.get(), allow_none=True).list()
            except (Fault, ProtocolError, OSError) as exc:
                return self._post(lambda: self.status.set(f"刷新失败：{exc}"))
            self._post(lambda: self._fill(names))
        threading.Thread(target=work, daemon=True).start()     # 网络不堵 UI

    def _fill(self, names: list[str]) -> None:
        self.box.delete(0, tk.END)
        for name in names:
            self.box.insert(tk.END, name)
        self.status.set(f"共 {len(names)} 个文件")

    def download(self) -> None:
        if not (selection := self.box.curselection()):
            return messagebox.showinfo("提示", "请先选中一个文件")
        name = self.box.get(selection[0])

        def work() -> None:
            try:
                data = ServerProxy(self.peer.get(), allow_none=True).fetch(name, SECRET)
            except Fault as exc:
                return self._post(lambda: messagebox.showerror(
                    "服务端", f"[{exc.faultCode}] {exc.faultString}"))
            except (ProtocolError, OSError) as exc:
                return self._post(lambda: messagebox.showerror("连接", str(exc)))
            OUTDIR.mkdir(exist_ok=True)
            dest = (OUTDIR / Path(name).name).resolve()
            if not dest.is_relative_to(OUTDIR):                # 落盘也要防穿越
                return self._post(lambda: messagebox.showerror("拒绝", "目标路径越界"))
            dest.write_bytes(data.data)
            self._post(lambda: self.status.set(f"已保存 {dest}"))

        threading.Thread(target=work, daemon=True).start()

if __name__ == "__main__":
    root = tk.Tk(); App(root); root.mainloop()
```

### 28.6 进一步探索与「2026 年等价做法」

| 方向 | 做法 |
| --- | --- |
| 进度反馈 | 下载改成分块 HTTP，用 `ttk.Progressbar` 显示百分比（XML-RPC 无法增量） |
| 换 UI 技术栈 | `textual`（终端 UI，异步友好）、PySide6/Qt（原生观感）、Web UI（FastAPI + 浏览器） |
| 打包与线程模型 | PyInstaller / Nuitka / `briefcase` 取代 `py2exe`；改用 `asyncio` 时可用 `asyncio.to_thread` |

| 原书环节 | 原书做法 | 2026 年等价做法 |
| --- | --- | --- |
| 界面库与线程 | `Tkinter`（Python 3 起全小写 `tkinter`）+ 手动 `threading` | 仍是 `tkinter`（维护中），或 `textual` / PySide6 / Web UI；并发改用 `asyncio` + `asyncio.to_thread` |
| 传输 | XML-RPC | HTTP(S) 分块下载（见第 27 章的等价做法） |

## 近年研究与工业界开源实践（2015–2026）

- **`textual` / Web UI**：前者在终端里做富交互界面，原生 `asyncio` 驱动，是本章 GUI 的最轻量现代替代；后者（FastAPI + 浏览器）把「装一个客户端」变成「打开一个网址」。
- **PySide6 / PyQt 与 `tkinter`**：前者原生观感与控件集完整，适合交付的桌面软件；后者标准库自带、无额外依赖，教学与内部小工具仍是合理选择（高 DPI、主题需额外打磨）。
- **打包分发与研究侧**：PyInstaller / Nuitka / `briefcase` 取代 `py2exe`，桌面分发还需处理签名与自动更新；HCI 研究对「异步进度反馈」「可取消操作」的结论一致——超过约 1 秒的操作都应给出进度与取消入口。🔧 各项目版本号与 API 细节以官方文档为准。

## 常见误区与本书需修正之处

| # | 原书写法 / 常见误区 | 问题 | 2026 年正确写法 |
| --- | --- | --- | --- |
| 1 | 🔧 沿用 Python 2 的 `import Tkinter` | 模块不存在 | Python 3 起为全小写 `tkinter` |
| 2 | 🔧 线程模型错误：在回调里做网络请求，或子线程直接操作控件 | 界面卡死「未响应」；跨线程调用 Tk 会随机崩溃 | 耗时操作进子线程，再用 `root.after(0, ...)` 回主线程更新 |
| 4 | 🔧 只对服务端校验路径 | 客户端把 `../../` 写进系统目录 | 落盘前同样 `resolve()` + `is_relative_to()` |
| 5 | 🔧 异常未捕获、服务器线程非 daemon | 一个 `ProtocolError` 直接弹栈终止；关窗口后进程不退 | 按 `Fault` / `ProtocolError` / `OSError` 分类提示；`Thread(daemon=True)` + `shutdown()` |
| 7 | 🔧 无进度反馈、地址硬编码 | 大文件下载时用户以为卡死；无法连别的节点 | 分块下载 + `ttk.Progressbar`；地址与密钥走输入/环境变量 |
| 8 | 🔧 用 `py2exe` 打包、或认为 GUI 只是「套一层皮」 | Windows-only 且已边缘化；真正的改动在线程模型与错误传播 | PyInstaller / Nuitka / `briefcase`；先定「谁在哪个线程」，再画控件 |
