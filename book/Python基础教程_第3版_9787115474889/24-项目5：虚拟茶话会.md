# 第 24 章 项目5：虚拟茶话会（原书 pp.364-378）

### 24.2 先看清原书工具链的去向

```python
# ❌ 历史对照，Python 3.13 上无法运行：asyncore/asynchat 已被 PEP 594 移除
import asyncore
import asynchat
import socket

class ChatSession(asynchat.async_chat):
    def __init__(self, server, sock):
        super().__init__(sock); self.server = server
        self.set_terminator(b"\r\n")        # 以行作为消息边界
        self.data = []

    def collect_incoming_data(self, data):  # 数据分片到达时累积
        self.data.append(data.decode("utf-8", "replace"))

    def found_terminator(self):             # 一行到齐后才分发
        line = "".join(self.data); self.data = []
        self.server.broadcast(line)
```

这套回调式 API（`collect_incoming_data` / `found_terminator`）是 `asyncio` 出现之前的经典形态。今天的标准答案是 **PEP 3156 的 `asyncio` + PEP 492 的 `async/await`**：把「等待一行数据」写成 `await reader.readline()`，回调地狱直接消失。

### 24.3–24.4 初次实现：asyncio 版 ChatServer / ChatSession / 整合

```python
# chat_asyncio.py —— 教学示意，不参与构建
# 纯标准库，Python 3.11+（用到 match 与 TaskGroup 风格写法）
# 运行：python chat_asyncio.py  然后另开终端：nc 127.0.0.1 8888
from __future__ import annotations

import asyncio
from dataclasses import dataclass, field

WELCOME = "欢迎来到虚拟茶话会。输入 /help 查看命令。"

@dataclass(eq=False)    # eq=False：保留按身份哈希，会话才能放进 set
class ChatSession:
    server: "ChatServer"
    reader: asyncio.StreamReader
    writer: asyncio.StreamWriter
    name: str = "匿名"
    rooms: set[str] = field(default_factory=lambda: {"Main"})

    async def send(self, msg: str) -> None:
        self.writer.write((msg + "\n").encode("utf-8"))
        await self.writer.drain()          # 背压：写太快时在这里等

    async def serve(self) -> None:
        await self.send(WELCOME)
        while line := await self.reader.readline():   # 海象运算符（PEP 572）
            text = line.decode("utf-8", "replace").strip()
            if not text:
                continue
            await self.server.handle(self, text)
        await self.server.disconnect(self)

class ChatServer:
    def __init__(self) -> None:
        self.sessions: set[ChatSession] = set()
        self.rooms: dict[str, set[ChatSession]] = {"Main": set()}

    async def handle(self, session: ChatSession, text: str) -> None:
        if text.startswith("/"):
            cmd, _, rest = text.partition(" ")
            await self.command(session, cmd.lower(), rest.strip())
        else:
            for room in session.rooms:
                await self.broadcast(f"[{room}] {session.name}: {text}", room, skip=session)

    async def command(self, session: ChatSession, cmd: str, rest: str) -> None:
        match cmd:                                  # 结构模式匹配（PEP 634/635/636）
            case "/login":
                session.name = rest or "匿名"
                await session.send(f"你好，{session.name}！")
            case "/join":
                await self.join(session, rest or "Main")
            case "/leave":
                await self.leave(session, rest or "Main")
            case "/rooms":
                await session.send("房间：" + "、".join(sorted(self.rooms)))
            case "/who":
                room = self.rooms.get(rest or "Main") or ()
                await session.send("在线：" + "、".join(s.name for s in room))
            case "/quit":
                await self.disconnect(session)
            case "/help":
                await session.send("命令：/login 名字 /join 房 /leave 房 /rooms /who 房 /quit")
            case _:
                await session.send(f"未知命令：{cmd}")

    async def join(self, session: ChatSession, room: str) -> None:
        self.rooms.setdefault(room, set()).add(session)
        session.rooms.add(room)
        await self.broadcast(f"*{session.name} 加入了 {room}*", room)

    async def leave(self, session: ChatSession, room: str) -> None:
        self.rooms.get(room, set()).discard(session)
        session.rooms.discard(room)
        await session.send(f"你已离开 {room}。")

    async def broadcast(self, msg: str, room: str, skip: ChatSession | None = None) -> None:
        targets = [s for s in self.rooms.get(room, ()) if s is not skip]
        for s in targets:                            # 逐连接写，慢连接不拖垮别人
            try:
                await s.send(msg)
            except (ConnectionResetError, BrokenPipeError):
                await self.disconnect(s)

    async def connect(self, reader: asyncio.StreamReader, writer: asyncio.StreamWriter) -> None:
        session = ChatSession(self, reader, writer)
        self.sessions.add(session); self.rooms["Main"].add(session)
        try:
            await session.serve()
        except (ConnectionResetError, asyncio.CancelledError):
            await self.disconnect(session)

    async def disconnect(self, session: ChatSession) -> None:
        if session in self.sessions:
            self.sessions.discard(session)
            for room in self.rooms.values():
                room.discard(session)
            for name in list(session.rooms):
                await self.broadcast(f"*{session.name} 离开了*", name)
            session.rooms.clear()
        if not session.writer.is_closing():
            session.writer.close(); await session.writer.wait_closed()

async def main() -> None:
    server = ChatServer()
    async with await asyncio.start_server(server.connect, "127.0.0.1", 8888) as srv:
        print("茶话会已启动：nc 127.0.0.1 8888")
        await srv.serve_forever()

if __name__ == "__main__":
    asyncio.run(main())          # asyncio.run：3.7+ 的统一入口（PEP 3156 生态）
```

三个设计点值得单独记住：**消息边界**从回调式 `set_terminator` 变成 `await reader.readline()`；**背压**从无（缓冲区无限增长）变成 `await writer.drain()`；**生命周期**从 `handle_close` 回调变成 `async with asyncio.start_server(...)`。

### 24.5 再次实现：命令解释、聊天室、登录退出、主聊天室

上面的代码已经把 24.5 的五个子目标一次性落地了（命令解释 / 聊天室 / 登录退出 / 主聊天室 / 新服务器）：

- **基本命令解释**：`/` 前缀识别 + `match` 分发，未知命令显式报错，而不是静默忽略。
- **聊天室**：`server.rooms: dict[str, set[ChatSession]]`，房间是「会话引用的集合」，不是会话的属性。
- **登录与退出**：`/login` 只改昵称；退出必须**同时**清 `sessions` 与所有 `rooms` 里的引用，否则会出现「幽灵成员」——消息发给一个已关闭的连接。
- **主聊天室**：`Main` 房默认存在且不可退出；**新的服务器**扩展命令时只动 `command()`，不动 I/O 层。

配套的脚本化客户端（本地自测，不需要 `nc`）：

```python
# chat_client.py —— 教学示意，不参与构建（需先启动 chat_asyncio.py）
import asyncio

async def main() -> None:
    reader, writer = await asyncio.open_connection("127.0.0.1", 8888)
    async def pump():
        while line := await reader.readline():
            print(">>", line.decode("utf-8", "replace").rstrip())
    task = asyncio.create_task(pump())
    for msg in ["/login 小明", "/join Python", "大家好！", "/who Python", "/quit"]:
        writer.write((msg + "\n").encode("utf-8"))
        await writer.drain()
        await asyncio.sleep(0.2)
    task.cancel(); writer.close(); await writer.wait_closed()

asyncio.run(main())
```

## 近年研究与工业界开源实践（2015–2026）

- **`asyncio` 成为事实标准**：3.4 引入后经 3.7–3.11 的数轮打磨（`asyncio.run`、`TaskGroup`、超时语义统一），长连接服务的默认选型已稳定。
- **`websockets` / `aiohttp`**：把 WebSocket 服务端与客户端做成 `asyncio` 原生协程，是本章聊天室通往浏览器的最短路径。
- **FastAPI（ASGI）**、**Redis Pub/Sub** 与 **`pytest-asyncio`**：分别解决「HTTP+WS 一体」「多实例房间升级为频道」「用内存流测协议不必真开端口」。
- **研究侧**：并发模型的研究重心已从回调/反应器转向**结构化并发**（任务树与取消传播）；Python 的 `TaskGroup` 是其在工业语言中的代表实现。🔧 各工具版本号与 API 细节以官方文档为准。

## 常见误区与本书需修正之处

| # | 原书写法 / 常见误区 | 问题 | 2026 年正确写法 |
| --- | --- | --- | --- |
| 1 | 🔴 用 `asyncore`/`asynchat` | 3.12 弃用、**3.13 移除（PEP 594）** | `asyncio`（PEP 3156）+ `async/await`（PEP 492） |
| 2 | 🔧 回调式 `found_terminator` 状态机 | 状态散落在实例属性里，异常路径难清理 | `await reader.readline()` 顺序写 |
| 3 | 🔧 广播时不处理慢/断连接 | 一个卡死的客户端会拖住整轮广播 | 逐连接 `await drain()` + 捕获 `ConnectionResetError` |
| 4 | 🔧 退出时只从 `sessions` 移除 | 房间里残留引用 → 给已关闭连接写数据 | 退出时同步清理 `sessions` 与所有 `rooms` |
| 5 | 🔧 完全没有鉴权，昵称随便填 | 冒名、刷屏、无审计 | 连接建立后限时完成鉴权，token 与会话绑定 |
| 6 | 🔧 用 `str.split()` 解析命令，无未知命令提示 | 拼错命令被静默吞掉 | `partition` + `match` 显式兜底 |
| 7 | 认为「异步 = 更快」或消息无长度上限 | 异步只省**等待**；超长行能打爆内存 | CPU 密集走进程池；读取后按字节截断 |
| 8 | 🔧 把 `print` 当服务器日志 | 无分级、无时间戳、多人并发时交错 | `logging`（PEP 282），按连接 ID 记录 |
| 9 | 🔧 认为 GIL 让多核无用从而放弃并发设计 | 3.13 起有 free-threaded 构建（PEP 703），3.14 官方支持 | I/O 密集用 `asyncio`；CPU 密集评估多进程/多解释器 |
