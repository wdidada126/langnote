import java.time.LocalDate;

public class Main {
    public static void main(String[] args) {
        // 假设我们要检查的日期是2024年10月28日
        LocalDate dateToCheck = LocalDate.of(2024, 10, 28);

        // 获取该日期所在月的第一天
        LocalDate firstDayOfMonth = dateToCheck.minusDays(dateToCheck.getDayOfMonth() - 1);

        // 判断是否是该月的第一天
        if (dateToCheck.equals(firstDayOfMonth)) {
            System.out.println(dateToCheck + " 是该月的第一天");
        } else {
            System.out.println(dateToCheck + " 不是该月的第一天");
        }
    }
