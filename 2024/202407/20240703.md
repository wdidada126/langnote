好的，这些概念是现代 C++ 开发的核心。下面我将为您详细列出每个概念的引入版本，并提供一个综合的 CMake 项目演示。

各概念推出的 C++ 版本

概念 引入版本 说明

STL (Standard Template Library) C++98 标准模板库（容器、算法、迭代器）是 C++ 最初标准化的核心部分。

指针 (Pointers) C++98 (及之前) 从 C 语言继承而来，是 C++ 最基础的概念。

模板 (Templates) C++98 泛型编程的基础，与 STL 一同成为 C++ 的核心特征。

std::bind / std::function C++11 实现了可调用对象的统一封装和参数绑定，是函数式编程的重要工具。

std::thread C++11 提供了标准化的多线程支持，可以跨平台创建和管理线程。

std::async / std::future / std::promise C++11 提供了基于任务的异步编程模型，可以更方便地获取异步操作的结果。

#pragma once

#include <functional>
#include <vector>
#include <future>
#include <memory>

// 演示 std::function 和 std::bind
class TaskScheduler {
public:
    using Task = std::function<int(int)>; 

    void addTask(Task task);
    std::vector<int> executeAll(int baseValue);

private:
    std::vector<Task> m_tasks;
};

// 演示模板函数
template <typename T>
T add(const T& a, const T& b) {
    return a + b;
}

// 演示智能指针和异步操作
class DataProcessor {
public:
    DataProcessor();
    std::future<std::vector<int>> processAsync(const std::vector<int>& data);
    
private:
    std::vector<int> heavyProcessing(const std::vector<int>& data);
};

// 演示多线程和 promise/future
std::future<int> executeTaskInThread(int param);


2. 实现文件 (src/utils.cpp)

#include "utils.h"
#include <iostream>
#include <thread>
#include <chrono>

void TaskScheduler::addTask(Task task) {
    m_tasks.push_back(task);
}

std::vector<int> TaskScheduler::executeAll(int baseValue) {
    std::vector<int> results;
    for (auto& task : m_tasks) {
        results.push_back(task(baseValue));
    }
    return results;
}

DataProcessor::DataProcessor() {
    std::cout << "DataProcessor created\n";
}

std::future<std::vector<int>> DataProcessor::processAsync(const std::vector<int>& data) {
    // 使用 async 启动异步任务
    return std::async(std::launch::async, [this, data]() {
        return heavyProcessing(data);
    });
}

std::vector<int> DataProcessor::heavyProcessing(const std::vector<int>& data) {
    std::vector<int> result;
    for (int x : data) {
        // 模拟耗时操作
        std::this_thread::sleep_for(std::chrono::milliseconds(10));
        result.push_back(x * 2 + 1);
    }
    return result;
}

std::future<int> executeTaskInThread(int param) {
    // 演示 promise/future 用于线程间通信
    auto promise = std::make_shared<std::promise<int>>();
    std::future<int> future = promise->get_future();
    
    std::thread([promise, param]() {
        try {
            // 模拟工作
            std::this_thread::sleep_for(std::chrono::milliseconds(100));
            int result = param * 10;
            promise->set_value(result);  // 设置结果
        } catch (...) {
            promise->set_exception(std::current_exception());
        }
    }).detach();  // 分离线程
    
    return future;
}


3. 主程序 (src/main.cpp)

#include "utils.h"
#include <iostream>
#include <memory>
#include <vector>

// 自由函数用于演示 bind
int multiply(int a, int b) {
    return a * b;
}

int main() {
    std::cout << "=== Modern C++ Features Demo ===\n\n";
    
    // 1. 演示 std::function 和 std::bind
    std::cout << "1. std::function and std::bind:\n";
    TaskScheduler scheduler;
    
    // 绑定自由函数
    scheduler.addTask(std::bind(multiply, std::placeholders::_1, 5));
    
    // 使用 lambda
    scheduler.addTask([](int x) { return x + 10; });
    
    auto results = scheduler.executeAll(20);
    for (size_t i = 0; i < results.size(); ++i) {
        std::cout << "  Task " << i + 1 << " result: " << results[i] << "\n";
    }
    
    // 2. 演示模板
    std::cout << "\n2. Templates:\n";
    std::cout << "  add(5, 3) = " << add(5, 3) << "\n";
    std::cout << "  add(2.5, 3.7) = " << add(2.5, 3.7) << "\n";
    
    // 3. 演示智能指针和 STL 容器
    std::cout << "\n3. Smart pointers and STL:\n";
    auto processor = std::make_unique<DataProcessor>();
    std::vector<int> inputData = {1, 2, 3, 4, 5};
    
    // 4. 演示 async/future
    auto futureResult = processor->processAsync(inputData);
    
    std::cout << "  Async processing started...\n";
    auto processedData = futureResult.get();  // 阻塞等待结果
    
    std::cout << "  Processed data: ";
    for (int val : processedData) {
        std::cout << val << " ";
    }
    std::cout << "\n";
    
    // 5. 演示 promise/future 与线程
    std::cout << "\n4. Thread with promise/future:\n";
    auto threadFuture = executeTaskInThread(42);
    std::cout << "  Thread task result: " << threadFuture.get() << "\n";
    
    std::cout << "\n=== Demo Completed ===\n";
    return 0;
}


4. CMake 配置文件 (CMakeLists.txt)
cmake_minimum_required(VERSION 3.15)  # 要求较新版本的 CMake
project(ModernCppDemo LANGUAGES CXX)  # 指定为 C++ 项目

# 设置 C++ 标准为 14（兼容 11 的所有特性）
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)   # 必须支持该标准
set(CMAKE_CXX_EXTENSIONS OFF)         # 禁用编译器扩展

# 设置包含目录
include_directories(include)

# 查找所有源文件
file(GLOB SOURCES "src/*.cpp")

# 创建可执行文件
add_executable(demo ${SOURCES})

# 设置编译器特性要求（C++11 线程支持）
target_compile_features(demo PRIVATE cxx_std_14)

# 在支持线程的系统上链接线程库
find_package(Threads REQUIRED)
target_link_libraries(demo Threads::Threads)

# 设置输出目录（可选）
set_target_properties(demo PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin
)

# 在 Windows 上需要定义这个宏来启用线程支持
if(WIN32)
    target_compile_definitions(demo PRIVATE _WIN32_WINNT=0x0601)
endif()

# 安装规则（可选）
install(TARGETS demo DESTINATION bin)
