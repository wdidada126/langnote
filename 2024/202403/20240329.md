CREATE
	VIEW MKT_CUST_INFO_TD_VIEW AS SELECT
		*
	FROM
		(
		SELECT
			MCIT.*,
			MOIT.SPEC_ID,
			MOIT.AMT,
			MOIT.ORDER_STAT,
			MOIT.ORDER_TIME
		FROM
			app_market_dev.MKT_CUST_INFO_TD MCIT
		LEFT JOIN app_market_dev.MKT_ORDER_INFO_TD MOIT ON
			MOIT.CUST_ID = MCIT.CUST_ID
		WHERE
			MCIT.DISMISS = '0'
			AND EXISTS (
			SELECT
				1
			FROM
				app_market_dev.MKT_CUST_PROD_INFO_TM MCPIT
			WHERE
				MCPIT.CUST_ID = MCIT.CUST_ID
				AND MCPIT.PROD_STAT = '1' )
			AND MCIT.TENA_ID = 'T20231027141354017540000000229'
		ORDER BY
			MOIT.ORDER_TIME DESC ) T
	GROUP BY
		T.CUST_ID;
