要在 CMakeLists.txt 中限制只能在 Windows 系统上编译，并在其他系统上报错，可以使用 `if` 和 `message` 命令来实现。以下是一个示例的 CMakeLists.txt 文件，它将在非 Windows 系统上显示一条错误信息并停止构建过程：

```cmake
cmake_minimum_required(VERSION 3.15)

project(MyProject)

# Check if the system is Windows
if(NOT WIN32)
    message(FATAL_ERROR "This project can only be compiled on Windows systems.")
endif()

# Rest of your CMake configuration goes here
# For example, you might set up targets, include directories, etc.

add_executable(MyExecutable main.cpp)
```

### 说明

1. cmake_minimum_required：设置 CMake 的最低版本要求。

2. project：定义项目名称。

3. if(NOT WIN32)：检查是否不是 Windows 系统。`WIN32` 是 CMake 内置的一个变量，当目标平台是 Windows 时，该变量被定义。如果不是 Windows 系统，`NOT WIN32` 将返回 `true`。

4. message(FATAL_ERROR "...")：如果条件为真，则使用 `message` 命令生成一个致命错误。这个错误会导致 CMake 过程停止。

5. add_executable：示例中，假设你有一个名为 `main.cpp` 的源文件，用于创建可执行文件。如果有其他配置项，可以在此位置添加。

这样，当你在非 Windows 系统上尝试编译时，CMake 会给出错误提示并停止配置过程，从而防止项目在不支持的系统上进行构建。

Windows I/O Completion Ports (IOCP) 是一种高效的异步 I/O 模型，通常用于开发高性能的网络服务器。下面是一个完整的使用 IOCP 来创建一个简单的 TCP 回显服务器的示例代码。这个服务器会接受客户端连接，并回显客户端发送的所有数据。

### 完整示例代码

主要部分：
- 初始化和配置网络套接字。
- 创建完成端口，并将套接字与完成端口关联。
- 启动一个线程池，等待来自完成端口的 I/O 操作完成。
- 处理客户端连接和数据回显。

```cpp
#define WIN32_LEAN_AND_MEAN
#include <windows.h>
#include <winsock2.h>
#include <ws2tcpip.h>
#include <mswsock.h>
#include <iostream>
#include <thread>
#include <vector>
#include <mutex>

#pragma comment(lib, "Ws2_32.lib")

constexpr int BUFFER_SIZE = 512;

struct PER_IO_OPERATION_DATA {
    OVERLAPPED overlapped;
    WSABUF dataBuffer;
    CHAR buffer[BUFFER_SIZE];
    SOCKET socket;
    DWORD bytesSent;
    DWORD bytesReceived;
    int operation; // 0 for receive, 1 for send
};

void workerThread(HANDLE completionPort) {
    DWORD bytesTransferred;
    ULONG_PTR completionKey;
    LPOVERLAPPED overlapped;

    while (true) {
        BOOL result = GetQueuedCompletionStatus(
            completionPort, &bytesTransferred, &completionKey, &overlapped, INFINITE);

        if (!result || bytesTransferred == 0) {
            // An error occurred or the client disconnected
            auto* perIoData = reinterpret_cast<PER_IO_OPERATION_DATA*>(overlapped);
            std::cout << "Client disconnected." << std::endl;
            closesocket(perIoData->socket);
            delete perIoData;
            continue;
        }

        auto* perIoData = reinterpret_cast<PER_IO_OPERATION_DATA*>(overlapped);
        SOCKET clientSocket = perIoData->socket;

        if (perIoData->operation == 0) { // Receive operation
            std::cout << "Received " << bytesTransferred << " bytes." << std::endl;
            perIoData->bytesReceived = bytesTransferred;
            perIoData->operation = 1; // Next operation is to send data back

            // Echo the data back to the client
            perIoData->dataBuffer.len = perIoData->bytesReceived;
            perIoData->dataBuffer.buf = perIoData->buffer;

            DWORD sendBytes;
            result = WSASend(
                clientSocket, &(perIoData->dataBuffer), 1, &sendBytes, 0, &(perIoData->overlapped), NULL);

            if (result == SOCKET_ERROR && WSAGetLastError() != WSA_IO_PENDING) {
                std::cout << "WSASend failed with error: " << WSAGetLastError() << std::endl;
                closesocket(clientSocket);
                delete perIoData;
                continue;
            }
        } else { // Send operation
            std::cout << "Sent " << bytesTransferred << " bytes." << std::endl;

            // Reset the operation data structure
            ZeroMemory(&(perIoData->overlapped), sizeof(OVERLAPPED));
            perIoData->bytesSent = 0;
            perIoData->bytesReceived = 0;
            perIoData->dataBuffer.len = BUFFER_SIZE;
            perIoData->dataBuffer.buf = perIoData->buffer;
            perIoData->operation = 0;

            DWORD flags = 0;
            result = WSARecv(
                clientSocket, &(perIoData->dataBuffer), 1, &bytesTransferred, &flags, &(perIoData->overlapped), NULL);

            if (result == SOCKET_ERROR && WSAGetLastError() != WSA_IO_PENDING) {
                std::cout << "WSARecv failed with error: " << WSAGetLastError() << std::endl;
                closesocket(clientSocket);
                delete perIoData;
                continue;
            }
        }
    }
}

int main() {
    WSADATA wsaData;
    int result = WSAStartup(MAKEWORD(2, 2), &wsaData);
    if (result != 0) {
        std::cout << "WSAStartup failed with error: " << result << std::endl;
        return 1;
    }

    SOCKET listenSocket = socket(AF_INET, SOCK_STREAM, IPPROTO_TCP);
    if (listenSocket == INVALID_SOCKET) {
        std::cout << "Socket failed with error: " << WSAGetLastError() << std::endl;
        WSACleanup();
        return 1;
    }

    sockaddr_in service;
    service.sin_family = AF_INET;
    service.sin_addr.s_addr = INADDR_ANY;
    service.sin_port = htons(27015);

    if (bind(listenSocket, (SOCKADDR*)&service, sizeof(service)) == SOCKET_ERROR) {
        std::cout << "Bind failed with error: " << WSAGetLastError() << std::endl;
        closesocket(listenSocket);
        WSACleanup();
        return 1;
    }

    if (listen(listenSocket, SOMAXCONN) == SOCKET_ERROR) {
        std::cout << "Listen failed with error: " << WSAGetLastError() << std::endl;
        closesocket(listenSocket);
        WSACleanup();
        return 1;
    }

    HANDLE completionPort = CreateIoCompletionPort(INVALID_HANDLE_VALUE, NULL, 0, 0);
    if (completionPort == NULL) {
        std::cout << "CreateIoCompletionPort failed with error: " << GetLastError() << std::endl;
        closesocket(listenSocket);
        WSACleanup();
        return 1;
    }

    // Create worker threads
    const int numThreads = std::thread::hardware_concurrency();
    std::vector<std::thread> threads;
    for (int i = 0; i < numThreads; ++i) {
        threads.emplace_back(workerThread, completionPort);
    }

    std::cout << "Server is listening on port 27015." << std::endl;

    while (true) {
        SOCKET clientSocket = accept(listenSocket, NULL, NULL);
        if (clientSocket == INVALID_SOCKET) {
            std::cout << "Accept failed with error: " << WSAGetLastError() << std::endl;
            continue;
        }

        // Associate the client socket with the completion port
        if (CreateIoCompletionPort((HANDLE)clientSocket, completionPort, 0, 0) == NULL) {
            std::cout << "CreateIoCompletionPort failed with error: " << GetLastError() << std::endl;
            closesocket(clientSocket);
            continue;
        }

        auto* perIoData = new PER_IO_OPERATION_DATA;
        ZeroMemory(&(perIoData->overlapped), sizeof(OVERLAPPED));
        perIoData->socket = clientSocket;
        perIoData->bytesSent = 0;
        perIoData->bytesReceived = 0;
        perIoData->dataBuffer.len = BUFFER_SIZE;
        perIoData->dataBuffer.buf = perIoData->buffer;
        perIoData->operation = 0;

        DWORD flags = 0;
        DWORD recvBytes;
        if (WSARecv(clientSocket, &(perIoData->dataBuffer), 1, &recvBytes, &flags, &(perIoData->overlapped), NULL) == SOCKET_ERROR) {
            if (WSAGetLastError() != WSA_IO_PENDING) {
                std::cout << "WSARecv failed with error: " << WSAGetLastError() << std::endl;
                closesocket(clientSocket);
                delete perIoData;
            }
        }
    }

    // Clean up
    for (auto& thread : threads) {
        thread.join();
    }

    closesocket(listenSocket);
    CloseHandle(completionPort);
    WSACleanup();

    return 0;
}
```

### 说明

1. 初始化网络：使用 `WSAStartup` 初始化 Winsock。
2. 创建监听套接字：设置监听套接字以接受客户端连接。
3. 创建完成端口：`CreateIoCompletionPort` 创建完成端口并将监听套接字与之关联。
4. 创建工作线程：使用标准库的 `std::thread` 创建一组工作线程，每个线程在循环中等待完成端口上的 I/O 完成。
5. 处理连接和 I/O：每个工作线程使用 `GetQueuedCompletionStatus` 等待 I/O 操作的完成，然后根据操作类型（接收或发送）执行相应的处理逻辑。
6. 接受新连接：主线程在循环中接受新的客户端连接，并将每个客户端套接字与完成端口关联。

### 注意事项

- 确保正确处理错误和清理资源，避免内存泄漏和套接字泄漏。
- 此示例代码没有处理并发连接或其他网络异常的边界情况。
- 根据实际需要，可以扩展此服务器的功能，例如增加多线程同步机制、超时处理、错误恢复等。

这个警告信息 C4996 是由 Microsoft Visual Studio 的编译器发出的，它指出 strerror 函数可能不安全，因为它返回一个指向静态存储区的指针，这个存储区可能会被后续的 strerror 调用覆盖。为了解决这个问题，建议使用更安全的函数 strerror_s。

修改方法
使用 strerror_s 替换 strerror
strerror_s 函数需要一个额外的参数来指定目标缓冲区的大小，这有助于防止缓冲区溢出。你可以按照以下方式修改代码：


```c
#include <string.h>  
#include <errno.h>  
 
char buf[256];  
int err = errno; // 假设你已经有了错误码  
strerror_s(buf, sizeof(buf), err);  
// 现在可以使用 buf 中的错误信息了
在你的例子中，假设 libeventmain.c 文件中某处使用了 strerror，你可以这样修改：

c
// 假设原代码是这样的  
const char* error_msg = strerror(errno);  
 
// 修改为  
char error_msg_buf[256];  
strerror_s(error_msg_buf, sizeof(error_msg_buf), errno);  
const char* error_msg = error_msg_buf;
```
禁用警告
如果你不想修改代码，或者出于某种原因需要使用 strerror，你可以通过定义 _CRT_SECURE_NO_WARNINGS 来禁用这个警告。这可以通过在源代码文件的顶部添加以下预处理指令来实现：

```c
#define _CRT_SECURE_NO_WARNINGS
```
或者，在 Visual Studio 的项目设置中，你可以添加这个宏到预处理器定义中。这通常在你的项目属性中设置，具体路径可能因 Visual Studio 的版本而异，但通常可以在“项目属性”->“C/C++”->“预处理器”->“预处理器定义”中找到。

使用 strerror_s 是更安全的做法，因为它避免了潜在的缓冲区溢出问题。
