这个依赖包是用于前端SM4加密数据的SM2解密驱动，具体来说是银联（UnionPay）或银联相关系统的密码解密组件。

一、包作用分析
com.ydx.cryptoPWD:com-ydx:1.0.0 核心功能：

1. SM2算法解密：解密前端通过SM4加密的数据
2. 密码解密驱动：专门处理支付/金融场景下的密码安全传输
3. 国密算法支持：SM2（非对称加密）+ SM4（对称加密）协同工作
4. 银联标准实现：符合中国银联的密码安全规范

三、核心使用示例

1. 基本解密流程

import com.ydx.cryptoPWD.SM2Decryptor;
import com.ydx.cryptoPWD.SM4Utils;

public class PasswordDecryptExample {
    
    // SM2私钥（Base64格式）
    private static final String SM2_PRIVATE_KEY = 
        "MHcCAQEEIBgBqQ2jG7V3W6t8vJZk7L8M9N2BvC1DxE4F5G6H7IoAoGCCqGSM49AwEHoUQDQgAE";
    
    public String decryptPassword(String encryptedData) {
        try {
            // 1. SM2解密获取SM4密钥
            SM2Decryptor sm2 = new SM2Decryptor(SM2_PRIVATE_KEY);
            String sm4KeyEncrypted = extractSM4Key(encryptedData);
            byte[] sm4Key = sm2.decrypt(sm4KeyEncrypted);
            
            // 2. SM4解密实际数据
            String cipherText = extractCipherText(encryptedData);
            SM4Utils sm4 = new SM4Utils(sm4Key);
            return sm4.decrypt(cipherText);
            
        } catch (Exception e) {
            throw new RuntimeException("密码解密失败", e);
        }
    }
    
    private String extractSM4Key(String encryptedData) {
        // 从加密数据中提取SM4密钥部分
        return encryptedData.substring(0, 128); // 示例
    }
    
    private String extractCipherText(String encryptedData) {
        // 从加密数据中提取密文部分
        return encryptedData.substring(128);
    }
}

2. 支付密码解密完整示例

import com.ydx.cryptoPWD.*;
import java.util.Base64;

/**
 * 银联支付密码解密组件
 */
public class UnionPayPasswordDecryptor {
    
    private final SM2KeyPair sm2KeyPair;
    private final CertificateManager certManager;
    
    public UnionPayPasswordDecryptor() {
        // 初始化SM2密钥对
        this.sm2KeyPair = new SM2KeyPair();
        
        // 加载银联证书
        this.certManager = new CertificateManager();
        certManager.loadCertificate("unionpay.cer");
    }
    
    /**
     * 解密前端加密的支付密码
     * 前端流程：密码 -> SM4加密 -> SM2加密SM4密钥 -> 传输
     */
    public String decryptPaymentPassword(EncryptedRequest request) {
        try {
            // 1. 验证签名
            if (!verifySignature(request)) {
                throw new SecurityException("签名验证失败");
            }
            
            // 2. SM2解密获取SM4密钥
            byte[] encryptedSm4Key = Base64.getDecoder()
                .decode(request.getEncryptedKey());
            byte[] sm4KeyBytes = sm2KeyPair.decrypt(encryptedSm4Key);
            
            // 3. 创建SM4解密器
            SM4Decryptor sm4 = new SM4Decryptor.Builder()
                .setKey(sm4KeyBytes)
                .setMode(SM4Decryptor.Mode.CBC)
                .setIv(request.getIv().getBytes())
                .build();
            
            // 4. SM4解密密码
            byte[] encryptedPassword = Base64.getDecoder()
                .decode(request.getEncryptedPassword());
            byte[] passwordBytes = sm4.decrypt(encryptedPassword);
            
            return new String(passwordBytes, "UTF-8");
            
        } catch (Exception e) {
            // 记录解密失败日志
            LogUtils.error("支付密码解密失败", e);
            throw new DecryptException("密码解密失败", e);
        }
    }
    
    /**
     * 生成响应（重新加密）
     */
    public EncryptedResponse encryptResponse(String plainText) {
        // 使用新的SM4密钥加密响应
        byte[] newSm4Key = KeyGenerator.generateSM4Key();
        SM4Encryptor sm4 = new SM4Encryptor(newSm4Key);
        
        byte[] encryptedData = sm4.encrypt(plainText.getBytes());
        byte[] encryptedKey = sm2KeyPair.encrypt(newSm4Key);
        
        return EncryptedResponse.builder()
            .encryptedData(Base64.getEncoder().encodeToString(encryptedData))
            .encryptedKey(Base64.getEncoder().encodeToString(encryptedKey))
            .timestamp(System.currentTimeMillis())
            .build();
    }
    
    private boolean verifySignature(EncryptedRequest request) {
        // 验证银联签名
        return certManager.verify(
            request.getDataToSign(),
            request.getSignature()
        );
    }
}


3. Spring Boot集成配置

# application.yml
ydx:
  crypto:
    sm2:
      private-key: ${SM2_PRIVATE_KEY:}
      public-key: ${SM2_PUBLIC_KEY:}
    sm4:
      mode: CBC
      padding: PKCS5Padding
    certificate:
      path: classpath:certs/
      unionpay: unionpay_prod.cer

@Configuration
public class CryptoConfig {
    
    @Value("${ydx.crypto.sm2.private-key}")
    private String sm2PrivateKey;
    
    @Value("${ydx.crypto.sm2.public-key}")
    private String sm2PublicKey;
    
    @Bean
    public SM2Decryptor sm2Decryptor() {
        return new SM2Decryptor(sm2PrivateKey);
    }
    
    @Bean
    public PasswordDecryptService passwordDecryptService() {
        return new PasswordDecryptService(sm2Decryptor());
    }
}

@Service
public class PaymentService {
    
    @Autowired
    private PasswordDecryptService decryptService;
    
    public PaymentResult processPayment(PaymentRequest request) {
        // 解密支付密码
        String password = decryptService.decrypt(
            request.getEncryptedPassword()
        );
        
        // 验证密码逻辑
        boolean valid = validatePassword(
            request.getUserId(), 
            password
        );
        
        if (valid) {
            return executePayment(request, password);
        } else {
            throw new InvalidPasswordException("密码错误");
        }
    }
}


4. 前端配合示例（Vue + JavaScript）

// 前端加密逻辑
import { SM2, SM4 } from 'sm-crypto'

class PasswordEncryptor {
  constructor(sm2PublicKey) {
    this.sm2PublicKey = sm2PublicKey
  }
  
  // 加密支付密码
  encryptPassword(password) {
    // 1. 生成随机的SM4密钥
    const sm4Key = this.generateSM4Key()
    
    // 2. SM4加密密码
    const encryptedPassword = SM4.encrypt(
      password, 
      sm4Key, 
      { mode: 'cbc', iv: '1234567890123456' }
    )
    
    // 3. SM2加密SM4密钥
    const encryptedKey = SM2.doEncrypt(
      sm4Key, 
      this.sm2PublicKey, 
      1 // C1C3C2格式
    )
    
    return {
      encryptedPassword,
      encryptedKey,
      timestamp: Date.now()
    }
  }
  
  generateSM4Key() {
    // 生成16字节随机密钥
    return Array.from({length: 16}, () => 
      Math.floor(Math.random() * 256)
    ).map(b => b.toString(16).padStart(2, '0')).join('')
  }
}

// 使用示例
const encryptor = new PasswordEncryptor('公钥字符串')
const encrypted = encryptor.encryptPassword('user123')
// 发送 encrypted 到后端

五、常见问题处理

1. 密钥管理：私钥必须安全存储（HSM或KMS）
2. 版本兼容：确保前后端使用相同的国密算法实现
3. 性能优化：SM2解密较慢，考虑缓存机制
4. 错误处理：完善的异常处理和日志记录
