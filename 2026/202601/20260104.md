二、启动 PostgreSQL 数据库的多种方式
方式1：使用 systemctl（推荐）
# 查看服务状态
sudo systemctl status postgresql
# 启动服务
sudo systemctl start postgresql
# 重启服务
sudo systemctl restart postgresql
# 停止服务
sudo systemctl stop postgresql
# 设置开机自启
sudo systemctl enable postgresql
# 查看服务日志
sudo journalctl -u postgresql -f

方式2：手动使用 pg_ctl
# 初始化数据库（如果还没初始化）
sudo -u postgres /usr/lib/postgresql/18/bin/initdb \
    -D /var/lib/postgresql/18/main \
    --encoding=UTF8 --locale=en_US.UTF-8

# 启动数据库
sudo -u postgres /usr/lib/postgresql/18/bin/pg_ctl \
    -D /var/lib/postgresql/18/main \
    -l /var/log/postgresql/postgresql-18-main.log start

# 查看状态
sudo -u postgres /usr/lib/postgresql/18/bin/pg_ctl \
    -D /var/lib/postgresql/18/main status

# 停止数据库
sudo -u postgres /usr/lib/postgresql/18/bin/pg_ctl \
    -D /var/lib/postgresql/18/main stop


方式3：使用服务脚本

# 使用 Ubuntu 的服务管理
sudo service postgresql start
sudo service postgresql stop
sudo service postgresql restart
sudo service postgresql status


三、完整的安装和启动脚本

#!/bin/bash
# PostgreSQL 18.1 完整安装和配置脚本
set -e

echo "=== PostgreSQL 18.1 安装配置 ==="

# 1. 添加官方仓库
echo "1. 添加 PostgreSQL 官方仓库..."
sudo sh -c 'echo "deb https://apt.postgresql.org/pub/repos/apt $(lsb_release -cs)-pgdg main" > /etc/apt/sources.list.d/pgdg.list'

# 导入 GPG 密钥
wget --quiet -O - https://www.postgresql.org/media/keys/ACCC4CF8.asc | sudo apt-key add -

# 2. 安装
echo "2. 安装 PostgreSQL 18..."
sudo apt-get update
sudo apt-get install -y \
    postgresql-18 \
    postgresql-client-18 \
    postgresql-contrib-18 \
    postgresql-18-pgvector

# 3. 查看版本
echo "3. 验证安装..."
/usr/lib/postgresql/18/bin/postgres --version
/usr/lib/postgresql/18/bin/psql --version

# 4. 启动服务
echo "4. 启动服务..."
sudo systemctl start postgresql
sudo systemctl enable postgresql

# 5. 配置
echo "5. 基本配置..."
# 修改监听地址
sudo sed -i "s/#listen_addresses = 'localhost'/listen_addresses = '*'/" \
    /etc/postgresql/18/main/postgresql.conf

# 修改认证方式
echo "host    all             all             0.0.0.0/0               md5" | \
    sudo tee -a /etc/postgresql/18/main/pg_hba.conf

# 6. 重启生效
echo "6. 重启服务..."
sudo systemctl restart postgresql

# 7. 设置 postgres 用户密码
echo "7. 设置 postgres 用户密码..."
sudo -u postgres psql -c "ALTER USER postgres WITH PASSWORD 'postgres';"
echo "PostgreSQL 18.1 安装完成！"

七、性能调优脚本
#!/bin/bash
# PostgreSQL 性能调优
CONFIG_FILE="/etc/postgresql/18/main/postgresql.conf"
echo "性能调优配置..."
# 备份原配置
sudo cp "$CONFIG_FILE" "${CONFIG_FILE}.backup.$(date +%Y%m%d)"
# 根据内存调整配置
TOTAL_MEM=$(free -g | awk '/^Mem:/{print $2}')
SHARED_BUFFERS=$((TOTAL_MEM * 256))MB
EFFECTIVE_CACHE_SIZE=$((TOTAL_MEM * 768))MB
WORK_MEM=16MB
MAINTENANCE_WORK_MEM=$((TOTAL_MEM * 64))MB

# 应用配置
sudo tee -a "$CONFIG_FILE" << EOF

# 性能优化配置
shared_buffers = $SHARED_BUFFERS
effective_cache_size = $EFFECTIVE_CACHE_SIZE
work_mem = $WORK_MEM
maintenance_work_mem = $MAINTENANCE_WORK_MEM
max_connections = 100
shared_preload_libraries = 'pg_stat_statements'
pg_stat_statements.max = 10000
pg_stat_statements.track = all
random_page_cost = 1.1
effective_io_concurrency = 200
max_wal_size = 2GB
min_wal_size = 1GB
checkpoint_completion_target = 0.9
wal_buffers = 16MB
default_statistics_target = 100
EOF

# 重启服务
sudo systemctl restart postgresql
echo "性能调优完成！"

九、安全配置

#!/bin/bash
# PostgreSQL 安全配置
echo "安全配置..."

# 1. 修改默认端口
sudo sed -i "s/port = 5432/port = 5433/" /etc/postgresql/18/main/postgresql.conf

# 2. 禁用不必要扩展
sudo -u postgres psql -c "DROP EXTENSION IF EXISTS dblink;"

# 3. 限制连接
sudo tee -a /etc/postgresql/18/main/pg_hba.conf << 'EOF'
# 只允许本地连接
host    all             all             127.0.0.1/32            md5
host    all             all             ::1/128                 md5
EOF

# 4. 启用 SSL
sudo openssl req -new -x509 -days 365 -nodes \
    -text -out /etc/ssl/certs/server.crt \
    -keyout /etc/ssl/private/server.key \
    -subj "/CN=localhost"

sudo chmod 600 /etc/ssl/private/server.key
sudo chown postgres:postgres /etc/ssl/private/server.key

# 配置 SSL
sudo tee -a /etc/postgresql/18/main/postgresql.conf << 'EOF'
ssl = on
ssl_cert_file = '/etc/ssl/certs/server.crt'
ssl_key_file = '/etc/ssl/private/server.key'
EOF

sudo systemctl restart postgresql
echo "安全配置完成！"

十、故障排除
常见问题解决

# 检查服务是否运行

sudo find /var/log/postgresql -name "*.log*" -mtime +7 -delete

# 4. 数据库启动失败
# 查看详细日志
sudo journalctl -u postgresql -xe
