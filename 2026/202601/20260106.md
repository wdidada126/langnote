我来为您提供完整的使用 Meson 生成 C++ 项目 Doxygen 文档的配置方案：

一、基本配置

1. 创建 meson.build 中的 Doxygen 配置

### meson.build
project('my-cpp-project', 'cpp',
  version: '1.0.0',
  license: 'MIT',
  default_options: [
    'cpp_std=c++17',
    'warning_level=3',
  ]
)

### 查找 Doxygen
doxygen = find_program('doxygen', required: false)

if doxygen.found()
  ### 创建 Doxygen 配置
  doxygen_conf = configuration_data()
  doxygen_conf.set('PROJECT_NAME', meson.project_name())
  doxygen_conf.set('PROJECT_NUMBER', meson.project_version())
  doxygen_conf.set('PROJECT_BRIEF', 'A C++ example project')
  doxygen_conf.set('OUTPUT_DIR', meson.current_build_dir() / 'docs')
  doxygen_conf.set('INPUT_DIR', meson.current_source_dir())
  doxygen_conf.set('EXAMPLE_PATH', meson.current_source_dir() / 'examples')
  doxygen_conf.set('IMAGE_PATH', meson.current_source_dir() / 'images')
  
  ### 生成 Doxyfile
  doxyfile = configure_file(
    input: 'Doxyfile.in',
    output: 'Doxyfile',
    configuration: doxygen_conf
  )
  
  ### 创建文档生成任务
  run_target('docs',
    command: [doxygen, doxyfile]
  )
  
  ### 或者作为自定义目标
  custom_target('generate-docs',
    input: doxyfile,
    output: 'html',
    command: [doxygen, doxyfile],
    build_by_default: false,
    install: false
  )
else
  warning('Doxygen not found, documentation will not be generated')
endif

三、高级配置方案

1. 增强的 Meson 配置

### meson.build - 增强版
project('my-cpp-project', 'cpp',
  version: '1.0.0',
  license: 'MIT',
  default_options: [
    'cpp_std=c++20',
    'buildtype=debugoptimized',
    'warning_level=3',
    'werror=false',
  ],
  meson_version: '>=0.60.0'
)

### 依赖检查
doxygen = find_program('doxygen', required: false)
dot = find_program('dot', required: false)  ### Graphviz

### 包含目录
incdirs = include_directories('include', 'src')

### 源文件
sources = files(
  'src/main.cpp',
  'src/utils.cpp',
  'src/parser.cpp',
  'include/utils.hpp',
  'include/parser.hpp',
)

### 生成可执行文件
executable('myapp',
  sources,
  include_directories: incdirs,
  cpp_args: ['-DDEBUG', '-DUSE_DOXYGEN'],
  dependencies: [],
  install: true
)

### Doxygen 文档生成
if doxygen.found()
  ### 配置数据
  doxygen_conf = configuration_data()
  doxygen_conf.set('PROJECT_NAME', meson.project_name())
  doxygen_conf.set('PROJECT_VERSION', meson.project_version())
  doxygen_conf.set('PROJECT_SOURCE_DIR', meson.current_source_dir())
  doxygen_conf.set('PROJECT_BUILD_DIR', meson.current_build_dir())
  doxygen_conf.set('HAVE_DOT', dot.found() ? 'YES' : 'NO')
  
  ### 读取版本信息
  version_info = vcs_tag(
    input: 'version.hpp.in',
    output: 'version.hpp',
    command: ['git', 'describe', '--always', '--tags'],
    fallback: meson.project_version()
  )
  
  ### 生成 Doxyfile
  doxyfile = configure_file(
    input: 'docs/Doxyfile.in',
    output: 'Doxyfile',
    configuration: doxygen_conf
  )
  
  ### 文档生成目标
  docs_target = custom_target('doxygen-docs',
    input: [doxyfile] + sources + [version_info],
    output: ['html', 'latex'],
    command: [doxygen, doxyfile],
    build_by_default: false,
    install: true,
    install_dir: get_option('datadir') / 'doc' / meson.project_name()
  )
  
  ### 运行目标（命令行使用）
  run_target('docs',
    command: [doxygen, doxyfile],
    depends: [version_info]
  )
  
  ### 安装后的额外处理
  install_data(
    'docs/readme.md',
    install_dir: get_option('datadir') / 'doc' / meson.project_name()
  )
  
  ### 创建卸载脚本
  configure_file(
    input: 'docs/uninstall.in',
    output: 'uninstall-docs',
    configuration: doxygen_conf
  )
else
  warning('Doxygen not found. Documentation will not be generated.')
endif

### 测试
test('basic', executable('test_basic', 'tests/test_basic.cpp',
  include_directories: incdirs,
  dependencies: []
))


2. 自动包含 Doxygen 注释检查

### 添加 Doxygen 注释检查
if doxygen.found()
  ### 创建自定义命令检查 Doxygen 注释
  doxygen_check = custom_target('doxygen-check',
    input: sources,
    output: 'doxygen_warnings.txt',
    command: [
      doxygen,
      '-',
      'EXTRACT_ALL=YES',
      'WARNINGS=YES',
      'WARN_IF_UNDOCUMENTED=YES',
      'WARN_IF_DOC_ERROR=YES',
      'WARN_NO_PARAMDOC=YES',
      'INPUT=' + ' '.join(sources),
      'QUIET=YES',
      'GENERATE_HTML=NO',
      'GENERATE_LATEX=NO',
    ],
    capture: true,
    build_by_default: false
  )
  
  ### 在构建时运行检查
  run_target('check-docs',
    command: ['cat', doxygen_check.full_path()]
  )
endif

五、使用示例

生成文档

### 配置项目
meson setup builddir
cd builddir

### 生成文档
meson compile docs
### 或
ninja docs

### 查看文档
open docs/html/index.html


在 CI/CD 中集成

### .github/workflows/docs.yml
name: Documentation
on:
  push:
    branches: [ main ]
  pull_request:
    branches: [ main ]

jobs:
  build-docs:
    runs-on: ubuntu-latest
    steps:
    - uses: actions/checkout@v3
    - name: Install dependencies
      run: |
        sudo apt-get update
        sudo apt-get install -y doxygen graphviz meson ninja-build
    - name: Configure
      run: meson setup build
    - name: Build documentation
      run: |
        cd build
        meson compile docs
    - name: Deploy to GitHub Pages
      uses: peaceiris/actions-gh-pages@v3
      with:
        github_token: ${{ secrets.GITHUB_TOKEN }}
        publish_dir: ./build/docs/html

      build_by_default: false
    )
  endif
endif

七、最佳实践

1. 使用 Doxygen 注释标准：
/**
 * @brief 简短描述
 * 
 * 详细描述
 * 
 * @param param1 参数1描述
 * @param param2 参数2描述
 * @return 返回值描述
 * @throws ExceptionType 异常描述
 */
int myFunction(int param1, float param2);


2. 版本控制集成：
### 自动从 Git 获取版本
version = vcs_tag(
  input: 'version.hpp.in',
  output: 'version.hpp',
  command: ['git', 'describe', '--tags', '--dirty'],
  replace_string: '@VERSION@'
)


3. 增量构建：
### 只处理更改的文件
run_target('docs-incremental',
  command: [
    doxygen,
    doxyfile,
    'EXCLUDE_PATTERNS=*_bak*',
    'RECURSIVE=YES'
  ],
  depends: sources
)


这样配置后，Meson 会自动管理 Doxygen 文档的生成，并与构建系统完美集成。
