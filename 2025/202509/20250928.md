import com.lowagie.text.Font;
import com.lowagie.text.Phrase;
import com.lowagie.text.Paragraph;
import com.lowagie.text.Element;
import com.lowagie.text.pdf.PdfPCell;
import java.text.DecimalFormat;

public class PdfUtils {
    // 原有方法修改
    public static PdfPCell getPdfCellNodisableLine(String showDate, Font textFont, int horizon) {
        // 将金额字符串转换为人民币大写和小写
        String amount = showDate; // 假设输入为 "1000000.00"
        String capitalAmount = convertToChineseAmount(amount); // 大写：壹佰万元整
        String lowercaseAmount = formatToLowercaseAmount(amount); // 小写：1,000,000.00元

        // 创建 Paragraph 用于布局大写和小写
        Paragraph paragraph = new Paragraph();
        
        // 大写金额（居左）
        Phrase capitalPhrase = new Phrase("人民币 (大写)：" + capitalAmount, textFont);
        capitalPhrase.setLeading(10f); // 设置行间距
        paragraph.add(capitalPhrase);

        // 添加空格以推小写到右侧
        paragraph.add(new Phrase("    ", textFont)); // 动态调整空格数量或使用表格布局更佳

        // 小写金额（居右）
        Phrase lowercasePhrase = new Phrase("小写：" + lowercaseAmount, textFont);
        paragraph.add(lowercasePhrase);

        // 创建 PdfPCell
        PdfPCell nameCell = new PdfPCell(paragraph);
        nameCell.disableBorderSide(0); // 保持原无边框设置
        nameCell.setHorizontalAlignment(horizon); // 保持原水平对齐（可设为 Element.ALIGN_LEFT）
        nameCell.setVerticalAlignment(Element.ALIGN_MIDDLE); // 上下居中
        nameCell.setFixedHeight(40); // 固定高度

        return nameCell;
    }

    // 人民币金额转大写
    public static String convertToChineseAmount(String amountStr) {
        try {
            double amount = Double.parseDouble(amountStr);
            String[] chineseDigits = {"零", "壹", "贰", "叁", "肆", "伍", "陆", "柒", "捌", "玖"};
            String[] units = {"", "拾", "佰", "仟", "万", "拾", "佰", "仟", "亿"};
            String[] decimalUnits = {"角", "分"};
            String yuan = "元";
            String whole = "整";

            // 整数部分处理
            long integerPart = (long) amount;
            String integerStr = String.valueOf(integerPart);
            StringBuilder chineseStr = new StringBuilder();

            boolean isZero = false;
            int zeroCount = 0;

            for (int i = 0; i < integerStr.length(); i++) {
                int digit = integerStr.charAt(i) - '0';
                int unitIndex = integerStr.length() - 1 - i;

                if (digit == 0) {
                    zeroCount++;
                    isZero = true;
                    if (zeroCount == 1 && unitIndex != 0) {
                        chineseStr.append(chineseDigits[0]);
                    }
                } else {
                    if (isZero) {
                        isZero = false;
                        zeroCount = 0;
                    }
                    chineseStr.append(chineseDigits[digit]).append(units[unitIndex]);
                }
            }

            // 小数部分处理
            double decimalPart = amount - integerPart;
            String decimalStr = String.format("%.2f", decimalPart).substring(2);
            StringBuilder decimalChinese = new StringBuilder();

            if (decimalPart == 0) {
                decimalChinese.append(whole);
            } else {
                for (int i = 0; i < decimalStr.length() && i < 2; i++) {
                    int digit = decimalStr.charAt(i) - '0';
                    if (digit != 0) {
                        decimalChinese.append(chineseDigits[digit]).append(decimalUnits[i]);
                    }
                }
            }

            // 拼接结果
            String result = chineseStr.toString() + yuan + decimalChinese.toString();
            // 处理特殊情况，如“零元整”
            if (integerPart == 0 && decimalPart == 0) {
                return "零元整";
            }
            return result.replaceAll("零+", "零").replaceAll("零元", "元");
        } catch (Exception e) {
            return "金额格式错误";
        }
    }

    // 格式化小写金额（带千位分隔符）
    public static String formatToLowercaseAmount(String amountStr) {
        try {
            double amount = Double.parseDouble(amountStr);
            DecimalFormat df = new DecimalFormat("#,##0.00");
            return df.format(amount) + "元";
        } catch (Exception e) {
            return "金额格式错误";
        }
    }
}

	1	大写转换（convertToChineseAmount）：
	◦	将数字金额（如 1000000.00）转换为中文大写（壹佰万元整）。
	◦	支持整数和小数部分（最多两位小数，角、分）。
	◦	处理特殊情况，如零金额、连续零等。
	2	小写格式化（formatToLowercaseAmount）：
	◦	使用 DecimalFormat 添加千位分隔符，格式为 1,000,000.00元。
	◦	确保金额显示为两位小数。
	3	单元格布局：
	◦	使用 Paragraph 组合 Phrase，拼接大写和小写金额。
	◦	通过空格（" "）简单实现大写居左、小写居右。建议改进：若需更精确对齐，可使用 PdfPTable 嵌套两个子单元格，一个居左、一个居右。
	4	保持原样式：
	◦	保留 disableBorderSide(0)、固定高度 40、垂直居中。
	◦	水平对齐仍使用传入的 horizon 参数，建议设为 Element.ALIGN_LEFT。
