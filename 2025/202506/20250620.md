你提到的完全正确：

>  C++ 标准库本身不直接支持多进程（如创建、管理子进程），但可以通过 操作系统提供的系统调用 实现：
> - Linux/Unix：`fork()` + `exec()` 系列函数
> - Windows：`CreateProcess()` 函数

下面是一个 跨平台 CMake 项目示例，演示如何在 C++ 中使用系统调用创建子进程，并通过 `main()` 函数展示基本用法。

## 项目结构

```
cpp-multiprocess-demo/
├── CMakeLists.txt
└── src/
    └── main.cpp
```

## 1. `CMakeLists.txt`

```cmake
cmake_minimum_required(VERSION 3.10)

project(MultiProcessDemo LANGUAGES CXX)

# 使用 C++17（无需更高版本，但确保兼容）
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# 设置构建类型
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Debug)
endif()

# 添加可执行文件
add_executable(multiprocess src/main.cpp)

# 设置输出目录
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
```

## 2. `src/main.cpp`

```cpp
#include <iostream>
#include <string>

#ifdef _WIN32
    #include <windows.h>
    #include <tchar.h>
    #include <stdio.h>
#else
    #include <unistd.h>
    #include <sys/wait.h>
    #include <cstdlib>
#endif

void demo_fork_exec() {
#ifdef _WIN32
    std::cout << "当前系统为 Windows，不支持 fork/exec 演示。\n";
#else
    std::cout << "\n=== Linux/Unix: fork + exec 示例 ===\n";

    pid_t pid = fork();

    if (pid < 0) {
        std::cerr << "fork 失败\n";
        return;
    }

    if (pid == 0) {
        // 子进程
        std::cout << "【子进程】PID: " << getpid() << "，正在执行...\n";
        // 使用 exec 替换当前进程为 /bin/echo
        execl("/bin/echo", "echo", "Hello from child process!", nullptr);
        // 如果 exec 成功，下面这行不会执行
        std::cerr << "execl 失败\n";
        exit(1);
    } else {
        // 父进程
        int status;
        waitpid(pid, &status, 0);  // 等待子进程结束
        if (WIFEXITED(status)) {
            std::cout << "【父进程】子进程 PID " << pid << " 已退出，返回码: " << WEXITSTATUS(status) << "\n";
        }
    }
#endif
}

void demo_create_process() {
#ifdef _WIN32
    std::cout << "\n=== Windows: CreateProcess 示例 ===\n";

    STARTUPINFO si;
    PROCESS_INFORMATION pi;

    ZeroMemory(&si, sizeof(si));
    si.cb = sizeof(si);
    ZeroMemory(&pi, sizeof(pi));

    // 创建记事本进程（示例）
    LPSTR cmd = "notepad.exe";  // 注意：必须是 LPSTR，不能是 const char*

    BOOL result = CreateProcess(
        nullptr,      // 可执行文件名（null 表示从命令行解析）
        cmd,          // 命令行（可修改为其他程序，如 calc.exe）
        nullptr,      // 进程安全属性
        nullptr,      // 线程安全属性
        FALSE,        // 是否继承句柄
        0,            // 创建标志
        nullptr,      // 环境变量（null 表示继承）
        nullptr,      // 当前工作目录
        &si,          // 启动信息
        &pi           // 进程信息（输出）
    );

    if (result) {
        std::cout << "【Windows】成功启动进程，PID: " << pi.dwProcessId << "\n";
        // 等待进程结束（可选）
        WaitForSingleObject(pi.hProcess, 5000); // 最多等待 5 秒
        CloseHandle(pi.hProcess);
        CloseHandle(pi.hThread);
    } else {
        std::cerr << "【Windows】CreateProcess 失败，错误码: " << GetLastError() << "\n";
    }
#else
    std::cout << "当前系统为 Linux，不支持 CreateProcess 演示。\n";
#endif
}

int main() {
    std::cout << "=== C++ 多进程系统调用演示 ===\n";
    std::cout << "操作系统: "
#ifdef _WIN32
              << "Windows"
#else
              << "Linux/Unix"
#endif
              << "\n";

    // 演示 fork + exec (Linux/Unix)
    demo_fork_exec();

    // 演示 CreateProcess (Windows)
    demo_create_process();

    std::cout << "\n=== 演示结束 ===\n";
    return 0;
}
```

##  构建和运行

### 在 Linux/macOS 上：

```bash

cmake ..

./bin/multiprocess

cd cpp-multiprocess-demo
mkdir build
cd build
cmake .. -G "Visual Studio 17 2022"  # 或使用 MinGW
cmake --build .
.\bin\multiprocess.exe
```

##  总结

| 特性 | Linux/Unix | Windows |
|||--|
| 创建进程 | `fork()` + `exec()` | `CreateProcess()` |
| 等待子进程 | `waitpid()` | `WaitForSingleObject()` |
| C++ 标准支持 | ❌ 否 | ❌ 否 |
| 跨平台性 | 差 | 差 |
| 推荐替代方案 | `std::system()`（简单命令） | 使用跨平台库（如 Boost.Process） |

## 💡 最佳实践建议

1. 简单命令：使用 `std::system("ls")` 或 `std::system("dir")`
2. 复杂控制：使用系统调用（如本例）
3. 跨平台项目：考虑使用 Boost.Process 库
4. 避免硬编码路径：使用环境变量或配置
5. 错误处理：始终检查返回值和错误码

> 🔥 核心结论：C++ 标准库不支持多进程，必须依赖操作系统 API。本示例展示了如何在 CMake 项目中组织跨平台多进程代码。
