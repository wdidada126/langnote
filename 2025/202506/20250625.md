writev()

#include <sys/uio.h>  // 必须包含这个头文件
ssize_t writev(int fd, const struct iovec *iov, int iovcnt);


#include <sys/uio.h>   // 必须！定义 writev 和 struct iovec

```c
#include <sys/uio.h>
#include <unistd.h>
#include <string.h>
#include <stdio.h>

int main() {
    const char *str1 = "Hello, ";
    const char *str2 = "world!\n";

    struct iovec iov[2];
    iov[0].iov_base = (void*)str1;
    iov[0].iov_len = strlen(str1);
    iov[1].iov_base = (void*)str2;
    iov[1].iov_len = strlen(str2);

    int connfd = STDOUT_FILENO; // 例如：写入到标准输出
    ssize_t ret = writev(connfd, iov, 2);

    if (ret == -1) {
        perror("writev error");
    }

    return 0;
}
```
