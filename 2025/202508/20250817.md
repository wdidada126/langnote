export default {
  data() {
    return {
      oldTopicName: '',
      newTopicName: '',
      newGroup: '',
      newCusSystem: '',
      newProSystem: '',
      newDesc: '',
      newInis: '',
      form: this.$form.createForm(this),
      isEditMode: false,
      drawerTitle: '创建Topic'
    };
  },
  props: {
    filterVisible: {
      type: Boolean,
      default: false,
    },
    element: {
      type: HTMLDivElement,
      default: null,
    },
    filterForm: {
      type: Object,
      default: () => ({}), 
    },
    editData: {
      type: Object,
      default: () => ({})
    }
  },
  watch: {
    filterVisible: {
      immediate: true,
      handler(newVal) {
        if (newVal) {
          this.$nextTick(() => {
            this.initFormData();
          });
        }
      }
    }
  },
  mounted() {
    if (this.editData && Object.keys(this.editData).length > 0) {
      this.isEditMode = true;
      this.oldTopicName = this.editData.topicName;
      this.drawerTitle = '修改Topic';
      this.$nextTick(() => {
        this.form.setFieldsValue({
          mode: 'edit',
          oldTopicName: this.editData.topicName,
          newTopicName: this.editData.topicName,
          newGroup: this.editData.group,
          newProSystem: this.editData.proSystem,
          newCusSystem: this.editData.cusSystem,
          newDesc: this.editData.tDesc,
          newInis: this.editData.inis,
        });
        this.newInis = this.editData.inis;
      });
    } else {
      this.isEditMode = false;
      this.drawerTitle = '创建Topic';
    }
  },
  methods: {
    initFormData() {
      if (this.editData && Object.keys(this.editData).length > 0) {
      // 处理 PRO_SYSTEM：过滤空字符串
      console.log(this.editData.proSystem )
        console.log('初始化表单数据:', this.editData);
        this.oldTopicName = this.editData.topicName || '';
        // 确保表单已初始化
        if (!this.form.getFieldDecorator) {
          console.warn('表单尚未初始化');
          return;
        }
        // 分批设置字段值
        setTimeout(() => {
          try {
            this.form.setFieldsValue({
              'newInis': { value: this.editData.inis },
              'newTopicName': this.editData.topicName || '',
              'oldTopicName': this.editData.topicName || '',
              'newGroup': this.editData.group || '',
              'newProSystem': this.editData.proSystem || [],
              'newCusSystem': this.editData.cusSystem || [],
              'newDesc': this.editData.tDesc || ''
            });
          } catch (error) {
            console.error('设置表单值出错:', error);
          }
        }, 100);
      }
    },
    triggerNode() {
      return this.element ? this.element : document.getElementById("app");
    },
    filterOption(input, option) {
      return (
        option.componentOptions.children[0].text
          .toLowerCase()
          .indexOf(input.toLowerCase()) >= 0
      );
    },
    // 取消
    handleCancel() {
      this.$emit("update:filterVisible", false);
    },
    // 确认
    handleSubmit() {
      this.form.validateFieldsAndScroll((err, values) => {
        if (!err) {
          const payload = {
            ...values,
            isEdit: this.isEditMode,
            originalTopic: values.oldTopicName || this.oldTopicName,
          };
          console.log('payload:', payload);
          this.$emit("callback", payload);
          this.$emit("update:filterVisible", false);
        }
      });
    },
    handleSubmit2() {
      this.form.validateFieldsAndScroll((err, values) => {
        if (!err) {
          this.$emit("callback", values);
          this.$emit("update:filterVisible", false);
        }
      });
    },
  },
};
</script>
