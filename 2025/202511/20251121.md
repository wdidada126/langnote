示例：Nginx 日志解析器

%{
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    char ip[16];
    char timestamp[32];
    char method[16];
    char url[256];
    int status;
    int response_size;
    char referer[256];
    char user_agent[512];
} nginx_log_t;

void process_log_entry(nginx_log_t* log);
%}

%option noyywrap

IP      ([0-9]{1,3}\.){3}[0-9]{1,3}
METHOD  (GET|POST|PUT|DELETE|HEAD|OPTIONS|PATCH)
STATUS  [0-9]{3}
SIZE    [0-9]+
DATE    \[[^\]]+\]
URL     [^ ]+
REFERER \"[^\"]*\"
UA      \"[^\"]*\"

%%

{IP}    {
    strncpy(current_log.ip, yytext, sizeof(current_log.ip)-1);
}

{DATE}  {
    strncpy(current_log.timestamp, yytext+1, sizeof(current_log.timestamp)-2);
    current_log.timestamp[strlen(current_log.timestamp)-1] = '\0';
}

{METHOD} {
    strncpy(current_log.method, yytext, sizeof(current_log.method)-1);
}

{URL}   {
    strncpy(current_log.url, yytext, sizeof(current_log.url)-1);
}

{STATUS} {
    current_log.status = atoi(yytext);
}

{SIZE}  {
    current_log.response_size = atoi(yytext);
}

{REFERER} {
    strncpy(current_log.referer, yytext+1, sizeof(current_log.referer)-2);
    current_log.referer[strlen(current_log.referer)-1] = '\0';
}

{UA}    {
    strncpy(current_log.user_agent, yytext+1, sizeof(current_log.user_agent)-2);
    current_log.user_agent[strlen(current_log.user_agent)-1] = '\0';
}

\n      {
    process_log_entry(&current_log);
    memset(&current_log, 0, sizeof(current_log));
}

.|\t|   ;

%%

nginx_log_t current_log;

void process_log_entry(nginx_log_t* log) {
    // 业务逻辑：统计、告警、存储等
    printf("IP: %s, Method: %s, URL: %s, Status: %d, Size: %d\n",
           log->ip, log->method, log->url, log->status, log->response_size);
    
    // 可以集成到监控系统、数据库存储等
}


2. 配置文件解析

应用场景

示例：INI 文件解析器

%{
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char current_section[64] = "";
char current_key[64] = "";
char current_value[256] = "";

void process_section(const char* section);
void process_key_value(const char* key, const char* value);
%}

%option noyywrap

SECTION     \[[a-zA-Z0-9_]+\]
KEY         [a-zA-Z_][a-zA-Z0-9_]*
VALUE       [^\n\r]*
COMMENT     ;.*$

%%

{SECTION}   {
    strncpy(current_section, yytext+1, sizeof(current_section)-2);
    current_section[strlen(current_section)-1] = '\0';
    process_section(current_section);
}

{KEY}[ \t]*=[ \t]*{VALUE} {
    char* equal_pos = strchr(yytext, '=');
    if (equal_pos) {
        int key_len = equal_pos - yytext;
        strncpy(current_key, yytext, key_len);
        current_key[key_len] = '\0';
        
        // 去除前后空格
        char* value_start = equal_pos + 1;
        while (*value_start == ' ' || *value_start == '\t') value_start++;
        
        strncpy(current_value, value_start, sizeof(current_value)-1);
        process_key_value(current_key, current_value);
    }
}

{COMMENT}   ;  // 忽略注释
[ \t\n\r]+  ;  // 忽略空白字符

%%

void process_section(const char* section) {
    printf("[Section] %s\n", section);
}

void process_key_value(const char* key, const char* value) {
    printf("  %s = %s\n", key, value);
}

示例：CSV 到 JSON 转换器

%{
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int is_first_field = 1;
int is_first_record = 1;

void start_record();
void end_record();
void process_field(const char* field);
void escape_json_string(const char* input);
%}

%option noyywrap

QUOTED_FIELD   \"([^\"]|\"\")*\" 
FIELD          [^,\n\r]*
COMMA          ,
NEWLINE        \n|\r\n?

%%

{QUOTED_FIELD}  {
    char* content = strdup(yytext+1);
    content[strlen(content)-1] = '\0';
    
    // 处理转义引号
    char* pos = content;
    while ((pos = strstr(pos, "\"\"")) != NULL) {
        memmove(pos, pos+1, strlen(pos));
    }
    
    process_field(content);
    free(content);
}

{FIELD}         {
    process_field(yytext);
}

{COMMA}         {
    printf(",");
    is_first_field = 0;
}

{NEWLINE}       {
    end_record();
    is_first_record = 0;
    is_first_field = 1;
}

%%

void start_record() {
    if (!is_first_record) {
        printf(",\n");
    }
    printf("  {");
    is_first_field = 1;
}

void end_record() {
    printf("}");
}

void process_field(const char* field) {
    if (is_first_field) {
        start_record();
    } else {
        printf(",");
    }
    
    printf("\"field%d\":\"", is_first_field ? 1 : 2); // 简化示例
    escape_json_string(field);
    printf("\"");
    
    is_first_field = 0;
}

void escape_json_string(const char* input) {
    for (const char* p = input; *p; p++) {
        switch (*p) {
            case '"': printf("\\\""); break;
            case '\\': printf("\\\\"); break;
            case '\b': printf("\\b"); break;
            case '\f': printf("\\f"); break;
            case '\n': printf("\\n"); break;
            case '\r': printf("\\r"); break;
            case '\t': printf("\\t"); break;
            default: printf("%c", *p); break;
        }
    }
}

int main() {
    printf("[\n");
    yylex();
    printf("\n]\n");
    return 0;
}

示例：HTTP 头解析器

%{
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

void process_http_method(const char* method);
void process_http_url(const char* url);
void process_http_version(const char* version);
void process_header_field(const char* field, const char* value);
%}

%option noyywrap

METHOD     (GET|POST|PUT|DELETE|HEAD|OPTIONS|PATCH|TRACE|CONNECT)
URL        [^ ]+
VERSION    HTTP/[0-9]\.[0-9]
HEADER     [a-zA-Z0-9-]+
VALUE      [^\r\n]*
CRLF       \r\n

%%

{METHOD}    { process_http_method(yytext); }
{URL}       { process_http_url(yytext); }
{VERSION}   { process_http_version(yytext); }

{HEADER}:[ \t]*{VALUE} {
    char* colon = strchr(yytext, ':');
    if (colon) {
        *colon = '\0';
        char* value = colon + 1;
        while (*value == ' ' || *value == '\t') value++;
        process_header_field(yytext, value);
    }
}

{CRLF}{CRLF} { 
    printf("End of headers\n");
    return 0; // 结束解析
}

{CRLF}      ; // 忽略单个CRLF

.           ; // 忽略其他字符

%%

void process_http_method(const char* method) {
    printf("HTTP Method: %s\n", method);
}

void process_http_url(const char* url) {
    printf("URL: %s\n", url);
}

void process_http_version(const char* version) {
    printf("HTTP Version: %s\n", version);
}

void process_header_field(const char* field, const char* value) {
    printf("Header: %s = %s\n", field, value);
}

示例：C 代码统计器

%{
#include <stdio.h>

int total_lines = 0;
int code_lines = 0;
int comment_lines = 0;
int blank_lines = 0;
int in_comment = 0;

void print_statistics();
%}

%option noyywrap

%x COMMENT

%%

"/*"            { 
    in_comment = 1; 
    comment_lines++; 
    BEGIN(COMMENT); 
}

<COMMENT>"*/"   { 
    in_comment = 0; 
    BEGIN(INITIAL); 
}

<COMMENT>\n     { comment_lines++; total_lines++; }
<COMMENT>.      ;

"//".*          { comment_lines++; total_lines++; }

\n              { 
    total_lines++; 
    if (in_comment) {
        comment_lines++;
    } else {
        blank_lines++;
    }
}

[ \t]+          ; // 忽略空格和制表符

.               { 
    if (!in_comment) {
        code_lines++;
    }
}

<<EOF>>         { 
    print_statistics(); 
    return 0; 
}

%%

void print_statistics() {
    printf("=== Code Statistics ===\n");
    printf("Total lines: %d\n", total_lines);
    printf("Code lines: %d\n", code_lines);
    printf("Comment lines: %d\n", comment_lines);
    printf("Blank lines: %d\n", blank_lines);
    printf("Comment ratio: %.1f%%\n", (float)comment_lines/total_lines*100);
}

在编译和链接过程中，"Undefined symbols"（未定义符号）和"Undefined reference"（未定义的引用）这两个术语经常被交替使用，但它们描述的是链接过程中不同阶段的问题。为了帮你快速把握核心区别，我先用一个表格来汇总它们的主要特征。

特征 Undefined symbols (未定义符号) Undefined reference (未定义的引用)
问题阶段 链接过程的符号解析阶段 链接过程的最终检查阶段
本质含义 链接器在处理所有输入文件后，其内部符号表中没有找到该符号的定义 一个符号被声明和引用了，但在提供的库或目标文件中找不到其实现

常见场景 1. 缺少链接库 (-l 选项)<br>2. 库的链接顺序不当<br>3. 符号在库中被隐藏（如使用static） 1. 函数/变量只有头文件声明，没有对应的源文件实现<br>2. 源文件未加入编译系统（如CMakeLists.txt）<br>3. C++名称修饰（Name Mangling）导致符号不匹配

错误示例 ld: symbol(s) not found for architecture x86_64 undefined reference to \function_name'`

排查与解决方法
当遇到这类错误时，可以按照以下思路进行排查：
1.  检查符号定义
    使用 nm 命令查看目标文件（.o）或库文件（.a/.so）中是否存在你需要的符号。注意，对于C++项目，编译器会对符号进行名称修饰（Name Mangling），可以使用 nm -C 选项来解码（demangle）符号，使其更易读，或者使用 c++filt 工具。
    # 查看目标文件中的符号
    nm -C your_object_file.o | grep your_symbol
    # 查看动态库中的符号
    nm -gDC your_library.so | grep your_symbol

2.  检查依赖关系
    确保所有必要的源文件都已加入编译（例如检查CMakeLists.txt），并且所有用到的库都已正确链接。特别注意库的链接顺序，依赖其他库的库应该放在前面。
3.  检查编译环境一致性
    确保编译器版本、C++ ABI设置等在整个项目中保持一致。特别是当项目涉及不同GCC版本编译的模块时，可能会因ABI不兼容（如GCC 4.x与5.x的std::string实现不同）而导致符号找不到。

希望这些解释和排查方法能帮助你更好地理解和解决在编译链接过程中遇到的符号定义问题。如果你遇到了具体的错误信息，可以提供更多细节，或许我能给出更具体的建议。
