%{
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    char ip[16];
    char timestamp[32];
    char method[16];
    char url[256];
    int status;
    int response_size;
    char referer[256];
    char user_agent[512];
} nginx_log_t;

void process_log_entry(nginx_log_t* log);
%}

%option noyywrap

IP      ([0-9]{1,3}\.){3}[0-9]{1,3}
METHOD  (GET|POST|PUT|DELETE|HEAD|OPTIONS|PATCH)
STATUS  [0-9]{3}
SIZE    [0-9]+
DATE    \[[^\]]+\]
URL     [^ ]+
REFERER \"[^\"]*\"
UA      \"[^\"]*\"

%%

{IP}    {
    strncpy(current_log.ip, yytext, sizeof(current_log.ip)-1);
}

{DATE}  {
    strncpy(current_log.timestamp, yytext+1, sizeof(current_log.timestamp)-2);
    current_log.timestamp[strlen(current_log.timestamp)-1] = '\0';
}

{METHOD} {
    strncpy(current_log.method, yytext, sizeof(current_log.method)-1);
}

{URL}   {
    strncpy(current_log.url, yytext, sizeof(current_log.url)-1);
}

{STATUS} {
    current_log.status = atoi(yytext);
}

{SIZE}  {
    current_log.response_size = atoi(yytext);
}

{REFERER} {
    strncpy(current_log.referer, yytext+1, sizeof(current_log.referer)-2);
    current_log.referer[strlen(current_log.referer)-1] = '\0';
}

{UA}    {
    strncpy(current_log.user_agent, yytext+1, sizeof(current_log.user_agent)-2);
    current_log.user_agent[strlen(current_log.user_agent)-1] = '\0';
}

\n      {
    process_log_entry(&current_log);
    memset(&current_log, 0, sizeof(current_log));
}

.|\t|   ;

%%

示例：INI 文件解析器

%{
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char current_section[64] = "";
char current_key[64] = "";
char current_value[256] = "";

void process_section(const char* section);
void process_key_value(const char* key, const char* value);
%}

%option noyywrap

SECTION     \[[a-zA-Z0-9_]+\]
KEY         [a-zA-Z_][a-zA-Z0-9_]*
VALUE       [^\n\r]*
COMMENT     ;.*$

%%

{SECTION}   {
    strncpy(current_section, yytext+1, sizeof(current_section)-2);
    current_section[strlen(current_section)-1] = '\0';
    process_section(current_section);
}

{KEY}[ \t]*=[ \t]*{VALUE} {
    char* equal_pos = strchr(yytext, '=');
    if (equal_pos) {
        int key_len = equal_pos - yytext;
        strncpy(current_key, yytext, key_len);
        current_key[key_len] = '\0';
        
        // 去除前后空格
        char* value_start = equal_pos + 1;
        while (*value_start == ' ' || *value_start == '\t') value_start++;
        
        strncpy(current_value, value_start, sizeof(current_value)-1);
        process_key_value(current_key, current_value);
    }
}

{COMMENT}   ;  // 忽略注释
[ \t\n\r]+  ;  // 忽略空白字符

%%

void process_section(const char* section) {
    printf("[Section] %s\n", section);
}

void process_key_value(const char* key, const char* value) {
    printf("  %s = %s\n", key, value);
}

示例：CSV 到 JSON 转换器

%{
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int is_first_field = 1;
int is_first_record = 1;

void start_record();
void end_record();
void process_field(const char* field);
void escape_json_string(const char* input);
%}

%option noyywrap

QUOTED_FIELD   \"([^\"]|\"\")*\" 
FIELD          [^,\n\r]*
COMMA          ,
NEWLINE        \n|\r\n?

%%

{QUOTED_FIELD}  {
    char* content = strdup(yytext+1);
    content[strlen(content)-1] = '\0';
    
    // 处理转义引号
    char* pos = content;
    while ((pos = strstr(pos, "\"\"")) != NULL) {
        memmove(pos, pos+1, strlen(pos));
    }
    
    process_field(content);
    free(content);
}

{FIELD}         {
    process_field(yytext);
}

{COMMA}         {
    printf(",");
    is_first_field = 0;
}

{NEWLINE}       {
    end_record();
    is_first_record = 0;
    is_first_field = 1;
}

%%

void start_record() {
    if (!is_first_record) {
        printf(",\n");
    }
    printf("  {");
    is_first_field = 1;
}

void end_record() {
    printf("}");
}

void process_field(const char* field) {
    if (is_first_field) {
        start_record();
    } else {
        printf(",");
    }
    
    printf("\"field%d\":\"", is_first_field ? 1 : 2); // 简化示例
    escape_json_string(field);
    printf("\"");
    
    is_first_field = 0;
}

void escape_json_string(const char* input) {
    for (const char* p = input; *p; p++) {
        switch (*p) {
            case '"': printf("\\\""); break;
            case '\\': printf("\\\\"); break;
            case '\b': printf("\\b"); break;
            case '\f': printf("\\f"); break;
            case '\n': printf("\\n"); break;
            case '\r': printf("\\r"); break;
            case '\t': printf("\\t"); break;
            default: printf("%c", *p); break;
        }
    }
}

int main() {
    printf("[\n");
    yylex();
    printf("\n]\n");
    return 0;
}

%{
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

void process_http_method(const char* method);
void process_http_url(const char* url);
void process_http_version(const char* version);
void process_header_field(const char* field, const char* value);
%}

%option noyywrap

METHOD     (GET|POST|PUT|DELETE|HEAD|OPTIONS|PATCH|TRACE|CONNECT)
URL        [^ ]+
VERSION    HTTP/[0-9]\.[0-9]
HEADER     [a-zA-Z0-9-]+
VALUE      [^\r\n]*
CRLF       \r\n

%%

{METHOD}    { process_http_method(yytext); }
{URL}       { process_http_url(yytext); }
{VERSION}   { process_http_version(yytext); }

{HEADER}:[ \t]*{VALUE} {
    char* colon = strchr(yytext, ':');
    if (colon) {
        *colon = '\0';
        char* value = colon + 1;
        while (*value == ' ' || *value == '\t') value++;
        process_header_field(yytext, value);
    }
}

{CRLF}{CRLF} { 
    printf("End of headers\n");
    return 0; // 结束解析
}

{CRLF}      ; // 忽略单个CRLF

.           ; // 忽略其他字符

%%

void process_http_method(const char* method) {
    printf("HTTP Method: %s\n", method);
}

void process_http_url(const char* url) {
    printf("URL: %s\n", url);
}

void process_http_version(const char* version) {
    printf("HTTP Version: %s\n", version);
}

void process_header_field(const char* field, const char* value) {
    printf("Header: %s = %s\n", field, value);
}

%{
#include <stdio.h>

int total_lines = 0;
int code_lines = 0;
int comment_lines = 0;
int blank_lines = 0;
int in_comment = 0;

void print_statistics();
%}

%option noyywrap

%x COMMENT

%%

"/*"            { 
    in_comment = 1; 
    comment_lines++; 
    BEGIN(COMMENT); 
}

<COMMENT>"*/"   { 
    in_comment = 0; 
    BEGIN(INITIAL); 
}

<COMMENT>\n     { comment_lines++; total_lines++; }
<COMMENT>.      ;

"//".*          { comment_lines++; total_lines++; }

\n              { 
    total_lines++; 
    if (in_comment) {
        comment_lines++;
    } else {
        blank_lines++;
    }
}

[ \t]+          ; // 忽略空格和制表符

.               { 
    if (!in_comment) {
        code_lines++;
    }
}

<<EOF>>         { 
    print_statistics(); 
    return 0; 
}

%%

void print_statistics() {
    printf("=== Code Statistics ===\n");
    printf("Total lines: %d\n", total_lines);
    printf("Code lines: %d\n", code_lines);
    printf("Comment lines: %d\n", comment_lines);
    printf("Blank lines: %d\n", blank_lines);
    printf("Comment ratio: %.1f%%\n", (float)comment_lines/total_lines*100);
}

Bison 有几种方法可以确定生成文件的文件夹。以下是完整的方法：

1. 使用 -o 选项（推荐）

# 指定输出文件路径和名称
bison -d -o /path/to/output/parser.tab.c input.y

# 这会生成：
# /path/to/output/parser.tab.c
# /path/to/output/parser.tab.h


2. 使用 --defines 选项

# 分别指定 .c 和 .h 文件的输出路径
bison -d -o /path/to/output/parser.c --defines=/path/to/output/parser.h input.y


3. 结合使用工作目录

# 切换到目标目录，然后运行 bison
cd /path/to/output/
bison -d /path/to/source/input.y

# 这会生成在当前目录：
# /path/to/output/input.tab.c
# /path/to/output/input.tab.h


4. 在 CMake 中的具体应用

方法1：使用 execute_process

# 设置输出目录
set(BISON_OUTPUT_DIR ${CMAKE_BINARY_DIR}/generated)

# 创建输出目录
file(MAKE_DIRECTORY ${BISON_OUTPUT_DIR})

# 运行 Bison
execute_process(COMMAND
    bison -d 
    -o ${BISON_OUTPUT_DIR}/calculator.tab.c
    ${CMAKE_CURRENT_SOURCE_DIR}/calculator.y
    WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
    RESULT_VARIABLE BISON_RESULT
)

if(NOT BISON_RESULT EQUAL 0)
    message(FATAL_ERROR "Bison failed to generate parser")
endif()

# 验证生成的文件
if(NOT EXISTS ${BISON_OUTPUT_DIR}/calculator.tab.c)
    message(FATAL_ERROR "Bison output file not found: ${BISON_OUTPUT_DIR}/calculator.tab.c")
endif()


方法2：使用 CMake 的 bison_target

find_package(BISON REQUIRED)

# 设置输出目录
set(BISON_OUTPUT_DIR ${CMAKE_BINARY_DIR}/generated)

3. CMake 对 ANTLR4 的支持

ANTLR4 在 CMake 中没有官方的内置支持，但可以通过以下方式集成：

方法1：使用 find_package（如果 ANTLR4 已安装）

# 查找 ANTLR4
find_package(antlr4-runtime REQUIRED)

# 自定义命令生成解析器
add_custom_command(
    OUTPUT 
        ${CMAKE_BINARY_DIR}/MyGrammarLexer.cpp
        ${CMAKE_BINARY_DIR}/MyGrammarParser.cpp
        ${CMAKE_BINARY_DIR}/MyGrammarBaseListener.cpp
    COMMAND 
        java -jar ${ANTLR4_JAR} -Dlanguage=Cpp 
        -o ${CMAKE_BINARY_DIR} 
        ${CMAKE_CURRENT_SOURCE_DIR}/MyGrammar.g4
    DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/MyGrammar.g4
    COMMENT "Generating ANTLR4 parser"
)

add_executable(my_program
    main.cpp
    ${CMAKE_BINARY_DIR}/MyGrammarLexer.cpp
    ${CMAKE_BINARY_DIR}/MyGrammarParser.cpp
)

target_include_directories(my_program PRIVATE ${CMAKE_BINARY_DIR})
target_link_libraries(my_program antlr4-runtime)


方法2：使用 FindANTLR.cmake 脚本

# 包含自定义的 FindANTLR 脚本
include(FindANTLR)

find_package(ANTLR REQUIRED)

antlr_target(
    MyGrammarParser
    ${CMAKE_CURRENT_SOURCE_DIR}/MyGrammar.g4
    OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}
    LANGUAGE Cpp
)

add_executable(my_program
    main.cpp
    ${ANTLR_MyGrammarParser_OUTPUTS}
)

target_link_libraries(my_program antlr4-runtime)

完整的 Bison + Flex 项目

cmake_minimum_required(VERSION 3.12)
project(Calculator LANGUAGES C)

# 设置 C 标准

set(CMAKE_C_STANDARD_REQUIRED ON)

    src/calculator.y
    ${GENERATED_DIR}/calculator.tab.c

# 生成词法分析器
flex_target(CalculatorLexer
    src/calculator.l
    ${GENERATED_DIR}/lex.yy.c

# 设置依赖关系
add_flex_bison_dependency(CalculatorLexer CalculatorParser)

# 创建可执行文件
add_executable(calculator
    src/main.c
    src/calculator_utils.c
    ${BISON_CalculatorParser_OUTPUTS}
    ${FLEX_CalculatorLexer_OUTPUTS}
)

# 包含目录和链接
target_include_directories(calculator PRIVATE 
    ${GENERATED_DIR}
    ${CMAKE_CURRENT_SOURCE_DIR}/include
)

cmake_minimum_required(VERSION 3.12)

# 设置 ANTLR4 JAR 路径

add_custom_command(

        ${CMAKE_BINARY_DIR}/ExprBaseListener.cpp
        ${CMAKE_BINARY_DIR}/ExprListener.cpp

        -o ${CMAKE_BINARY_DIR} 
        ${CMAKE_CURRENT_SOURCE_DIR}/grammar/Expr.g4
    DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/grammar/Expr.g4
    COMMENT "Generating ANTLR4 C++ parser"
)

add_executable(antlr_example
    src/main.cpp
    src/MyListener.cpp
    ${CMAKE_BINARY_DIR}/ExprLexer.cpp

    ${CMAKE_BINARY_DIR}/ExprBaseListener.cpp

总结
• Bison 和 Flex 有官方的 CMake 支持，使用 bison_target() 和 flex_target()
• ANTLR4 需要手动配置生成命令
• 推荐使用 CMake 的内置函数，因为它们自动处理依赖关系和重新生成
• 对于复杂的项目，考虑将生成的文件放在单独的目录中以便管理

使用这些函数可以大大简化构建系统的配置，并确保正确的依赖管理。
