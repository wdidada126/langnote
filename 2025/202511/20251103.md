以下是 Java 17 生成12位随机数的几种方式：
方法1：使用 Random 类（基本随机数）
```
import java.util.Random;

public class Random12Digit {
    public static void main(String[] args) {
        // 生成12位随机数（可能以0开头）
        Random random = new Random();
        long random12Digit = random.nextLong(900_000_000_000L) + 100_000_000_000L;
        System.out.println("12位随机数: " + random12Digit);
        
        // 或者使用字符串格式确保12位
        String random12DigitStr = String.format("%012d", Math.abs(random.nextLong()) % 1_000_000_000_000L);
        System.out.println("12位随机数(字符串): " + random12DigitStr);
    }
}
```

方法2：使用 ThreadLocalRandom（推荐，性能更好）
```
import java.util.concurrent.ThreadLocalRandom;

public class Random12Digit {
    public static void main(String[] args) {
        // 生成不以0开头的12位随机数
        long randomNumber = ThreadLocalRandom.current().nextLong(100_000_000_000L, 1_000_000_000_000L);
        System.out.println("12位随机数: " + randomNumber);
        
        // 批量生成
        for (int i = 0; i < 5; i++) {
            long num = ThreadLocalRandom.current().nextLong(100_000_000_000L, 1_000_000_000_000L);
            System.out.println("随机数 " + (i+1) + ": " + num);
        }
    }
}
```

方法3：使用 SecureRandom（加密安全）
```
import java.security.SecureRandom;

public class Random12Digit {
    public static void main(String[] args) {
        SecureRandom secureRandom = new SecureRandom();
        
        // 方法3.1：使用数值范围
        long secureRandomNum = secureRandom.nextLong(900_000_000_000L) + 100_000_000_000L;
        System.out.println("安全12位随机数: " + secureRandomNum);
        
        // 方法3.2：使用字符串拼接（确保12位）
        StringBuilder sb = new StringBuilder(12);
        for (int i = 0; i < 12; i++) {
            sb.append(secureRandom.nextInt(10)); // 0-9的随机数
        }
        String secureRandomStr = sb.toString();
        System.out.println("安全12位随机字符串: " + secureRandomStr);
    }
}
```

方法4：实用工具类
```
import java.security.SecureRandom;
import java.util.concurrent.ThreadLocalRandom;

public class RandomGenerator {
    
    /**
     * 生成12位随机数（数值类型）
     */
    public static long generate12DigitNumber() {
        return ThreadLocalRandom.current().nextLong(100_000_000_000L, 1_000_000_000_000L);
    }
    
    /**
     * 生成12位随机数字字符串（可能以0开头）
     */
    public static String generate12DigitString() {
        SecureRandom random = new SecureRandom();
        StringBuilder sb = new StringBuilder(12);
        for (int i = 0; i < 12; i++) {
            sb.append(random.nextInt(10));
        }
        return sb.toString();
    }
    
    /**
     * 生成不以0开头的12位随机数字字符串
     */
    public static String generate12DigitStringNoZeroStart() {
        SecureRandom random = new SecureRandom();
        StringBuilder sb = new StringBuilder(12);
        // 第一位：1-9
        sb.append(random.nextInt(9) + 1);
        // 剩余11位：0-9
        for (int i = 0; i < 11; i++) {
            sb.append(random.nextInt(10));
        }
        return sb.toString();
    }

    public static void main(String[] args) {
        System.out.println("12位随机数: " + generate12DigitNumber());
        System.out.println("12位随机字符串: " + generate12DigitString());
        System.out.println("12位随机字符串(不以0开头): " + generate12DigitStringNoZeroStart());
        
        // 生成10个示例
        System.out.println("\n生成10个12位随机数:");
        for (int i = 0; i < 10; i++) {
            System.out.println(generate12DigitStringNoZeroStart());
        }
    }
}
```

方法5：Java 17 新特性 - 使用 RandomGenerator API
```
import java.util.random.RandomGenerator;

public class Random12DigitJava17 {
    public static void main(String[] args) {
        RandomGenerator generator = RandomGenerator.getDefault();
        
        // 生成12位随机数
        long randomNum = generator.nextLong(100_000_000_000L, 1_000_000_000_000L);
        System.out.println("使用RandomGenerator: " + randomNum);
        
        // 或者使用新的ints()方法
        String randomStr = generator.ints(12, 0, 10)
                .collect(StringBuilder::new, StringBuilder::append, StringBuilder::append)
                .toString();
        System.out.println("12位随机字符串: " + randomStr);
    }
}
```

推荐使用：
1. 对于一般需求：使用方法2的 ThreadLocalRandom，性能最好
2. 对于安全敏感场景（如订单号、验证码）：使用方法3的 SecureRandom
3. 需要字符串格式：使用方法4的工具类
注意：12位随机数的范围是100,000,000,000 到 999,999,999,999（如果不以0开头），总共9000亿种可能。
