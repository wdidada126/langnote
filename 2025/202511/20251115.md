您说得对！问题在于缺少 .pc (pkg-config) 和 .cmake (CMake 配置文件) 文件。这些文件对于库的自动发现至关重要。让我提供针对 CentOS 7 的解决方案：

问题分析

CentOS 7 的包管理确实缺少很多开发文件的配置：
• .pc 文件：pkg-config 用于查找库的编译和链接参数

• .cmake 文件：CMake 用于自动查找库的配置文件

解决方案

方法1：手动指定库路径（最直接）

cd /tmp/tmp.kELYPgeEOp/playscript-cpp

# 手动指定所有库路径，绕过 pkg-config 和 CMake 的自动查找
/home/wdidada/cmake-3.27.1-linux-x86_64/bin/cmake \
  -DCMAKE_BUILD_TYPE=Debug \
  -DCMAKE_MAKE_PROGRAM=/usr/bin/make \
  -DCMAKE_C_COMPILER=clang \
  -DCMAKE_CXX_COMPILER=clang++ \
  -DCMAKE_CXX_FLAGS="-std=c++11 -I/usr/include -I/usr/local/include" \
  -DCMAKE_EXE_LINKER_FLAGS="-L/usr/lib64 -L/usr/local/lib" \
  -DLLVM_DIR=/usr/lib64/llvm7.0/lib/cmake/llvm \
  -DLLVM_USE_STATIC_LIBS=OFF \
  -DANTLR4_RUNTIME_INCLUDE_DIRS=/tmp/tmp.kELYPgeEOp/playscript-cpp/antlr4.7.2-runtime \
  -DANTLR4_RUNTIME_LIBRARIES="" \
  -G "Unix Makefiles" \
  -S . \
  -B cmake-build-debug-aliyun_wdidada


方法2：创建缺失的 .pc 文件

# 为 LLVM 创建 pkg-config 文件
sudo mkdir -p /usr/lib64/pkgconfig

# 创建 llvm.pc 文件
sudo tee /usr/lib64/pkgconfig/llvm.pc << 'EOF'
prefix=/usr/lib64/llvm7.0
exec_prefix=${prefix}
libdir=${prefix}/lib
includedir=${prefix}/include

Name: LLVM
Description: LLVM compiler framework
Version: 7.0
Libs: -L${libdir} -lLLVM
Cflags: -I${includedir}
EOF

# 为系统库创建基本的 .pc 文件
sudo tee /usr/lib64/pkgconfig/antlr4-runtime.pc << 'EOF'
prefix=/usr/local
exec_prefix=${prefix}
libdir=${exec_prefix}/lib
includedir=${exec_prefix}/include

Name: antlr4-runtime
Description: ANTLR4 C++ Runtime
Version: 4.7.2
Libs: -L${libdir} -lantlr4-runtime
Cflags: -I${includedir}
EOF


方法3：创建缺失的 .cmake 文件

# 创建 ANTLR4 的 CMake 配置文件
sudo mkdir -p /usr/lib64/cmake/antlr4-runtime

sudo tee /usr/lib64/cmake/antlr4-runtime/antlr4-runtime-config.cmake << 'EOF'
# ANTLR4 Runtime CMake configuration file

set(ANTLR4_RUNTIME_FOUND TRUE)
set(ANTLR4_RUNTIME_INCLUDE_DIRS "/tmp/tmp.kELYPgeEOp/playscript-cpp/antlr4.7.2-runtime")
set(ANTLR4_RUNTIME_LIBRARIES "")

add_library(antlr4-runtime INTERFACE IMPORTED)
set_target_properties(antlr4-runtime PROPERTIES
  INTERFACE_INCLUDE_DIRECTORIES "${ANTLR4_RUNTIME_INCLUDE_DIRS}"
)

message(STATUS "Found ANTLR4 Runtime: ${ANTLR4_RUNTIME_INCLUDE_DIRS}")
EOF


方法4：修改项目的 CMakeLists.txt 直接包含

# 备份原 CMakeLists.txt
cp /tmp/tmp.kELYPgeEOp/playscript-cpp/CMakeLists.txt /tmp/tmp.kELYPgeEOp/playscript-cpp/CMakeLists.txt.backup

# 修改 CMakeLists.txt，直接指定路径而不是使用 find_package
cat > /tmp/tmp.kELYPgeEOp/playscript-cpp/CMakeLists.txt << 'EOF'
cmake_minimum_required(VERSION 3.10)
project(PlayScript)

# 直接设置包含路径，绕过 find_package
set(ANTLR4_RUNTIME_INCLUDE_DIRS "${CMAKE_SOURCE_DIR}/antlr4.7.2-runtime")
include_directories(${ANTLR4_RUNTIME_INCLUDE_DIRS})

# 设置 LLVM 路径
set(LLVM_DIR "/usr/lib64/llvm7.0/lib/cmake/llvm")

# 查找 LLVM（这个应该有 .cmake 文件）
find_package(LLVM 7.0 REQUIRED CONFIG)

# 添加编译定义
add_definitions(-DANTLR4CPP_NO_UNICODE=1)
add_definitions(-std=c++11)

# 您的其他配置...
message(STATUS "ANTLR4 include dir: ${ANTLR4_RUNTIME_INCLUDE_DIRS}")

# 继续原有的配置...
EOF

推荐的完整解决方案

# 1. 设置环境变量
export CMAKE_PREFIX_PATH=/usr/lib64/llvm7.0:$CMAKE_PREFIX_PATH
export PKG_CONFIG_PATH=/usr/lib64/pkgconfig:$PKG_CONFIG_PATH

# 2. 创建必要的配置文件
sudo mkdir -p /usr/lib64/pkgconfig
sudo tee /usr/lib64/pkgconfig/basic.pc << 'EOF'
prefix=/usr
exec_prefix=${prefix}
libdir=${prefix}/lib64
includedir=${prefix}/include

Name: Basic
Description: Basic system libraries
Version: 1.0
Libs: -L${libdir}
Cflags: -I${includedir}
EOF

# 3. 直接运行 CMake，手动指定路径
cd /tmp/tmp.kELYPgeEOp/playscript-cpp
/home/wdidada/cmake-3.27.1-linux-x86_64/bin/cmake \
  -DCMAKE_BUILD_TYPE=Debug \
  -DCMAKE_CXX_FLAGS="-std=c++11 -I${PWD}/antlr4.7.2-runtime" \
  -DLLVM_DIR=/usr/lib64/llvm7.0/lib/cmake/llvm \
  -G "Unix Makefiles" \
  -S . \
  -B cmake-build-debug-aliyun_wdidada
