LLVM Pass 核心概念详解

1. 什么是 LLVM Pass？

LLVM Pass 是 LLVM 编译器的基本构建块，用于对 LLVM 中间表示（IR）进行转换、分析和优化的模块化组件。

核心特点：

• 模块化：每个 Pass 完成特定功能

• 可组合：可以按顺序组合多个 Pass

• 可重用：可以在不同优化级别重复使用

• 可扩展：用户可以自定义 Pass

2. Pass 的主要类型

2.1 分析 Pass（Analysis Passes）

目的：收集程序信息，但不修改 IR
#include "llvm/IR/Function.h"
#include "llvm/Pass.h"
#include "llvm/Support/raw_ostream.h"

using namespace llvm;

namespace {
    // 基本块计数分析 Pass
    struct BasicBlockCounter : public FunctionPass {
        static char ID;
        BasicBlockCounter() : FunctionPass(ID) {}
        
        bool runOnFunction(Function &F) override {
            errs() << "Function: " << F.getName() << "\n";
            errs() << "Basic Blocks: " << F.size() << "\n";
            return false; // 不修改函数
        }
    };
}

char BasicBlockCounter::ID = 0;
static RegisterPass<BasicBlockCounter> X("bb-counter", "Count basic blocks");


2.2 转换 Pass（Transformation Passes）

目的：修改 IR 以优化程序
#include "llvm/IR/Function.h"
#include "llvm/IR/Instructions.h"
#include "llvm/Pass.h"

using namespace llvm;

namespace {
    // 简单的死代码消除 Pass
    struct DeadCodeElimination : public FunctionPass {
        static char ID;
        DeadCodeElimination() : FunctionPass(ID) {}
        
        bool runOnFunction(Function &F) override {
            bool changed = false;
            for (auto &BB : F) {
                for (auto it = BB.begin(); it != BB.end(); ) {
                    Instruction *I = &*it++;
                    if (isInstructionTriviallyDead(I)) {
                        I->eraseFromParent();
                        changed = true;
                    }
                }
            }
            return changed;
        }
    };
}

char DeadCodeElimination::ID = 0;
static RegisterPass<DeadCodeElimination> X("dce", "Dead Code Elimination");


2.3 模块 Pass（Module Passes）

目的：处理整个模块（多个函数）

#include "llvm/Analysis/LoopInfo.h"
#include "llvm/Pass.h"

namespace {
    class LoopAnalyzer : public FunctionPass {
    public:
        static char ID;
        LoopAnalyzer() : FunctionPass(ID) {}
        
        void getAnalysisUsage(AnalysisUsage &AU) const override {
            AU.addRequired<LoopInfoWrapperPass>(); // 需要循环信息

7.1 常量折叠 Pass
#include "llvm/IR/Constants.h"
#include "llvm/IR/Instruction.h"
#include "llvm/IR/Instructions.h"
#include "llvm/Pass.h"

namespace {
    class SimpleConstantFolder : public FunctionPass {
    public:
        static char ID;
        SimpleConstantFolder() : FunctionPass(ID) {}
        
        bool runOnFunction(Function &F) override {
            bool changed = false;
            
            for (BasicBlock &BB : F) {
                for (auto it = BB.begin(); it != BB.end(); ) {
                    Instruction *I = &*it++;
                    
                    if (auto *BO = dyn_cast<BinaryOperator>(I)) {
                        // 尝试常量折叠
                        if (Constant *C = ConstantFoldBinaryOpOperands(
                                BO->getOpcode(), 
                                BO->getOperand(0), 
                                BO->getOperand(1), 
                                F.getParent()->getDataLayout())) {
                            BO->replaceAllUsesWith(C);
                            BO->eraseFromParent();
                            changed = true;
                        }
                    }
                }
            }
            return changed;
        }
    };
}

SSA（Static Single Assignment）形式详解

1. 什么是 SSA？

SSA（Static Single Assignment） 是一种中间表示形式，其中每个变量只被赋值一次，并且在使用前必须被定义。

核心特征：

• 每个变量只赋值一次

• 使用前必须定义

• 使用 φ（phi）函数处理控制流合并

2. 为什么需要 SSA？

2.1 传统代码的问题

// 传统代码示例
int x = 1;
if (condition) {
    x = 2;  // 同一个变量被多次赋值
} else {
    x = 3;  // 同一个变量被多次赋值
}
int y = x + 1;  // 难以确定 x 的值


2.2 SSA 形式的优势

// SSA 形式
int x1 = 1;
if (condition) {
    int x2 = 2;
} else {
    int x3 = 3;
}
int x4 = φ(x2, x3);  // φ 函数选择正确的值
int y = x4 + 1;


3. SSA 的核心概念

3.1 φ（phi）函数

φ 函数在控制流合并点插入，根据前驱基本块选择正确的值。
// 控制流图示例
        [entry]
          |
          v
        [if condition]
         /      \
        /        \
 [then branch]  [else branch]
        \        /
         \      /
        [merge point]


在 merge point 插入 φ 函数：
; LLVM IR 示例
entry:
  %x1 = add i32 1, 0
  br i1 %condition, label %then, label %else

then:
  %x2 = add i32 2, 0
  br label %merge

else:
  %x3 = add i32 3, 0
  br label %merge

merge:
  %x4 = phi i32 [%x2, %then], [%x3, %else]
  %y = add i32 %x4, 1


4. SSA 构造算法

4.1 基本步骤

1. 插入 φ 函数
2. 重命名变量

4.2 支配关系（Dominance）

SSA 构造依赖于支配关系分析：

• 支配者（Dominator）：如果从入口到 B 的所有路径都经过 A，则 A 支配 B

• 直接支配者（Immediate Dominator）：最接近的支配者

• 支配边界（Dominance Frontier）

4.3 支配边界计算

// 支配边界算法伪代码
for each basic block B in CFG:
    DF[B] = ∅
    for each predecessor P of B:
        runner = P
        while runner != idom(B):
            DF[runner] = DF[runner] ∪ {B}
            runner = idom(runner)


5. 经典的 SSA 构造算法

5.1 Cytron 等人的算法

// 算法步骤
1. 计算支配关系
2. 计算支配边界
3. 确定需要插入 φ 函数的位置
4. 插入 φ 函数
5. 重命名变量


5.2 算法实现示例

#include "llvm/IR/Dominators.h"
#include "llvm/IR/Function.h"
#include "llvm/IR/Instructions.h"

class SSAContructor {
public:
    void constructSSA(Function &F) {
        // 1. 计算支配关系
        DominatorTree DT;
        DT.recalculate(F);
        
        // 2. 计算支配边界
        DominanceFrontier DF;
        DF.analyze(DT);
        
        // 3. 插入 φ 函数
        insertPhiFunctions(F, DF);
        
        // 4. 重命名变量
        renameVariables(F, DT);
    }
    
private:
    void insertPhiFunctions(Function &F, DominanceFrontier &DF) {
        // 为每个变量确定需要插入 φ 函数的位置
        for (auto &BB : F) {
            for (auto &I : BB) {
                if (auto *Alloca = dyn_cast<AllocaInst>(&I)) {
                    // 处理分配指令
                    processVariable(Alloca, DF);
                }
            }
        }
    }
    
    void renameVariables(Function &F, DominatorTree &DT) {
        // 使用栈进行变量重命名
        std::map<Value*, std::stack<Value*>> variableStacks;
        
        // 深度优先遍历支配树
        for (auto *BB : depth_first(DT.getRootNode())) {
            renameInBlock(BB->getBlock(), variableStacks);
        }
    }
};


6. LLVM 中的 SSA 实现

6.1 LLVM IR 的 SSA 特性

; LLVM IR 天然支持 SSA
define i32 @example(i32 %a, i32 %b) {
entry:
  %sum = add i32 %a, %b      ; %sum 只被赋值一次
  %cmp = icmp sgt i32 %sum, 0
  br i1 %cmp, label %if.then, label %if.else

if.then:
  %result1 = mul i32 %sum, 2 ; %result1 只被赋值一次
  br label %merge

if.else:
  %result2 = add i32 %sum, 1 ; %result2 只被赋值一次
  br label %merge

merge:
  %result = phi i32 [%result1, %if.then], [%result2, %if.else]
  ret i32 %result
}


6.2 LLVM 的 SSA 构造器

#include "llvm/Transforms/Utils/SSAUpdater.h"
#include "llvm/IR/Instructions.h"

void promoteToSSA(Function &F) {
    SSAUpdater SSA;
    
    for (auto &BB : F) {
        for (auto &I : BB) {
            if (auto *PHI = dyn_cast<PHINode>(&I)) {
                // 处理 φ 节点
                processPHINode(PHI, SSA);
            }
        }
    }
}


7. SSA 的优化优势

7.1 简化数据流分析

2.2 更完整的示例

import org.llvm.*;
import java.util.*;

public class SimpleCompiler {
    private ContextRef context;
    private ModuleRef module;
    private IRBuilderRef builder;
    
    public SimpleCompiler() {
        LLVM.initialize();
        this.context = ContextRef.create();
        this.module = ModuleRef.create("SimpleModule", context);
        this.builder = IRBuilderRef.create(context);
    }
    
    public FunctionRef createAddFunction() {
        // 创建函数类型: int add(int a, int b)
        TypeRef int32 = TypeRef.int32Type(context);
        TypeRef[] paramTypes = {int32, int32};
        FunctionTypeRef funcType = FunctionTypeRef.create(int32, paramTypes, false);
        
        // 创建函数
        FunctionRef addFunc = module.addFunction("add", funcType);
        addFunc.setLinkage(Linkage.External);
        
        // 创建基本块
        BasicBlockRef entry = addFunc.appendBasicBlock("entry");
        builder.positionAtEnd(entry);
        
        // 获取参数
        ValueRef a = addFunc.getParam(0);
        ValueRef b = addFunc.getParam(1);
        
        // 生成加法指令
        ValueRef result = builder.buildAdd(a, b, "addresult");
        
        // 返回结果
        builder.buildRet(result);
        
        return addFunc;
    }
    
    public void compileToFile(String filename) {
        // 验证模块
        if (module.verify()) {
            System.out.println("Module verification failed!");
            return;
        }
        
        // 输出 IR
        String ir = module.printToString();
        System.out.println("Generated LLVM IR:");
        System.out.println(ir);
        
        try {
            // 写入文件
            Files.write(Paths.get(filename), ir.getBytes());
            System.out.println("IR written to: " + filename);
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
    
    public static void main(String[] args) {
        SimpleCompiler compiler = new SimpleCompiler();
        compiler.createAddFunction();
        compiler.compileToFile("output.ll");
    }
}

总结

CMakePresets.json 主要是手动编辑的，但：

创建方式优先级：

1. 手动创建（最常用、最灵活）
2. IDE 辅助（Visual Studio、CLion）
3. 从现有配置迁移（脚本、命令行）
4. 工具生成（有限功能）

推荐工作流：

1. 手动创建基础模板
2. 根据项目需求逐步完善
3. 使用继承减少重复配置
4. 定期验证和测试预设

对于新项目，建议从简单的预设开始，随着项目复杂度的增加逐步完善配置。
