FuncSymbol与符号表的关系
1.符号表
符号表是编译器中的一个全局数据结构，用于存储程序中所有标识符的信息，包括：
• 变量名、函数名、类名等
• 类型信息
• 作用域信息
• 内存地址或偏移量

2. FuncSymbol
FuncSymbol通常是符号表中专门用于管理函数信息的一个条目或子系统。

class FuncSymbol : public Symbol {
public:
    enum Linkage {
        EXTERNAL,    // 外部链接
        INTERNAL,    // 内部链接
        STATIC       // 静态链接
    };

private:
    Type* returnType;
    std::vector<std::pair<std::string, Type*>> parameters;
    Linkage linkage;
    bool isDefined;
    int frameSize;

public:
    FuncSymbol(const std::string& name, Type* retType, Linkage link = EXTERNAL)
        : Symbol(name, SYMBOL_FUNCTION), returnType(retType), linkage(link), 
          isDefined(false), frameSize(0) {}
    
    void addParameter(const std::string& name, Type* type) {
        parameters.push_back({name, type});
    }
    
    bool matchesCall(const std::vector<Type*>& argTypes) const {
        if (parameters.size() != argTypes.size()) return false;
        for (size_t i = 0; i < parameters.size(); ++i) {
            if (!parameters[i].second->isCompatibleWith(argTypes[i])) {
                return false;
            }
        }
        return true;
    }

};

• 在编译的各个阶段被使用
• 帮助实现类型检查、重载解析、代码生成等功能

1. configure 脚本的基本结构
最小化的 configure 脚本示例：

#!/bin/bash
# -*- coding: utf-8 -*-
#
# configure - 配置脚本示例

# 初始化变量
prefix="/usr/local"
exec_prefix="${prefix}"
bindir="${exec_prefix}/bin"

# 帮助信息
show_help() {
    cat << EOF
用法: ./configure [选项] [变量=值]

选项:
  --help              显示此帮助信息
  --prefix=PREFIX     安装目录前缀 [默认: /usr/local]
  --bindir=DIR        可执行文件目录 [PREFIX/bin]
EOF
}

# 参数解析
while [ $# -gt 0 ]; do
    case "$1" in
        --prefix=*)
            prefix="${1#*=}"
            shift
            ;;
        --bindir=*)
            bindir="${1#*=}"
            shift
            ;;
        --help)
            show_help
            exit 0
            ;;
        *)
            echo "错误: 未知选项 $1"
            show_help
            exit 1
            ;;
    esac
done

# 系统检测
check_system() {
    echo "检查系统环境..."
    
    # 检查编译器
    if ! command -v gcc >/dev/null 2>&1; then
        echo "错误: 未找到 gcc 编译器"
        exit 1
    fi
    
    # 检查 make
    if ! command -v make >/dev/null 2>&1; then
        echo "错误: 未找到 make 工具"
        exit 1
    fi
}

# 生成 config.h
generate_config_header() {
    cat > config.h << EOF
/* config.h - 由 configure 脚本生成 */
#ifndef CONFIG_H
#define CONFIG_H

/* 安装路径 */
#define PREFIX "${prefix}"
#define BINDIR "${bindir}"

/* 系统特性检测 */
$(if uname -s | grep -q Linux; then echo "#define HAVE_LINUX 1"; fi)
$(if command -v pthread_create >/dev/null 2>&1; then echo "#define HAVE_PTHREAD 1"; fi)

#endif /* CONFIG_H */
EOF
}

# 生成 Makefile
generate_makefile() {
    cat > Makefile << EOF
# Makefile - 由 configure 脚本生成

CC = gcc
CFLAGS = -Wall -O2
PREFIX = ${prefix}
BINDIR = ${bindir}

SRCS = main.c util.c
OBJS = \$(SRCS:.c=.o)
TARGET = myapp

all: \$(TARGET)

\$(TARGET): \$(OBJS)
	\$(CC) \$(CFLAGS) -o \$@ \$(OBJS)

%.o: %.c
	\$(CC) \$(CFLAGS) -c -o \$@ $<

install: \$(TARGET)
	install -d \$(BINDIR)
	install \$(TARGET) \$(BINDIR)

clean:
	rm -f \$(OBJS) \$(TARGET)

.PHONY: all install clean
EOF
}

# 主函数
main() {
    echo "配置 MyApp 1.0"
    
    check_system
    generate_config_header
    generate_makefile
    
    echo "配置完成!"
    echo "安装前缀: ${prefix}"
    echo "运行 'make' 来编译程序"
    echo "运行 'make install' 来安装程序"
}

# 运行主函数
main "$@"


2. 完整的 configure.ac 示例（Autotools 标准方式）

实际上，标准的做法是编写 configure.ac，然后通过 autoconf 生成 configure 脚本。

configure.ac：

# -*- Autoconf -*-
# Process this file with autoconf to produce a configure script.

AC_PREREQ([2.69])
AC_INIT([myapp], [1.0], [bug-report@example.com])
AC_CONFIG_SRCDIR([src/main.c])
AC_CONFIG_HEADERS([config.h])
AC_CONFIG_AUX_DIR([build-aux])

AM_INIT_AUTOMAKE([foreign subdir-objects])
AC_CONFIG_FILES([Makefile src/Makefile])

# 检查系统类型
AC_CANONICAL_HOST

# 检查程序
AC_PROG_CC
AC_PROG_INSTALL
AC_PROG_MAKE_SET
AC_PROG_RANLIB

# 检查编译器特性
AC_HEADER_STDC
AC_CHECK_HEADERS([stdlib.h string.h unistd.h])

# 检查库函数
AC_CHECK_FUNCS([malloc memset strdup])

# 检查库
AC_CHECK_LIB([m], [sqrt])
AC_CHECK_LIB([pthread], [pthread_create])

# 用户选项
AC_ARG_WITH([readline],
    [AS_HELP_STRING([--with-readline],
        [支持 readline 库 @<:@默认检查@:>@])],
    [],
    [with_readline=check])

AS_IF([test "x$with_readline" != xno],
    [AC_CHECK_LIB([readline], [readline], [
        AC_DEFINE([HAVE_READLINE], [1], [Define if you have readline])
        READLINE_LIBS="-lreadline"
    ], [
        AS_IF([test "x$with_readline" != xcheck],
            [AC_MSG_ERROR([readline 库未找到但被请求])])
    ])])

AC_SUBST([READLINE_LIBS])

# 输出文件
AC_CONFIG_FILES([Makefile src/Makefile])
AC_OUTPUT

echo "
MyApp 配置摘要:

  版本:        ${PACKAGE_VERSION}
  安装前缀:    ${prefix}
  编译器:      ${CC}
  编译器标志:  ${CFLAGS}

  特性:
    Readline 支持: ${with_readline}

现在运行 'make' 来构建程序
运行 'make install' 来安装程序
"

4. 手动编写功能完整的configure脚本

#!/bin/bash
# configure - 功能完整的手动配置脚本

# 严格的错误处理
set -euo pipefail

# 脚本信息
PACKAGE_NAME="myapp"
PACKAGE_VERSION="1.0"
PACKAGE_BUGREPORT="bug-report@example.com"

# 默认配置
prefix="/usr/local"
exec_prefix="${prefix}"
bindir="${exec_prefix}/bin"
sysconfdir="${prefix}/etc"
libdir="${exec_prefix}/lib"
includedir="${prefix}/include"
datarootdir="${prefix}/share"
mandir="${datarootdir}/man"

CC="gcc"
CFLAGS="-Wall -O2"
LDFLAGS=""
LIBS=""
enable_debug="no"
with_readline="check"

# 日志函数
log_info() { echo "configure: $*"; }
log_error() { echo "configure: 错误: $*" >&2; }
log_check() { echo "检查 $1..."; }

# 显示帮助
show_help() {
    cat << EOF
用法: $0 [选项] [变量=值]

安装目录:
  --prefix=PREFIX        主安装目录 [/usr/local]
  --exec-prefix=EPREFIX  架构相关文件目录 [PREFIX]
  --bindir=DIR           用户可执行文件目录 [EPREFIX/bin]

编译器选项:
  --enable-debug         启用调试模式 [no]
  --with-readline        启用 readline 支持 [check]

环境变量:
  CC=C编译器              C 编译器命令 [gcc]
  CFLAGS=C编译器标志      C 编译器标志
  LDFLAGS=链接器标志      链接器标志
  LIBS=链接库             要链接的库

示例:
  $0 --prefix=/usr --enable-debug
  CC=clang $0 --with-readline=yes
EOF
}

# 参数解析
parse_arguments() {
    while [ $# -gt 0 ]; do
        case "$1" in
            --prefix=*) prefix="${1#*=}"; shift ;;
            --bindir=*) bindir="${1#*=}"; shift ;;
            --enable-debug) enable_debug="yes"; shift ;;
            --with-readline=*) with_readline="${1#*=}"; shift ;;
            --help) show_help; exit 0 ;;
            CC=*) CC="${1#*=}"; shift ;;
            CFLAGS=*) CFLAGS="${1#*=}"; shift ;;
            LDFLAGS=*) LDFLAGS="${1#*=}"; shift ;;
            LIBS=*) LIBS="${1#*=}"; shift ;;
            *) log_error "未知选项: $1"; show_help; exit 1 ;;
        esac
    done
}

# 系统检测
check_system() {
    log_info "执行系统检查"
    
    # 检查编译器
    log_check "C 编译器"
    if ! command -v "$CC" >/dev/null 2>&1; then
        log_error "未找到 C 编译器: $CC"
        exit 1
    fi
    
    # 测试编译器
    if ! "$CC" --version >/dev/null 2>&1; then
        log_error "编译器 $CC 无法执行"
        exit 1
    fi
}

# 检查头文件
check_header() {
    local header="$1"
    local tmpfile="/tmp/conftest.$$"
    
    cat > "$tmpfile.c" << EOF
#include <$header>
int main() { return 0; }
EOF
    
    if "$CC" $CFLAGS -o "$tmpfile" "$tmpfile.c" >/dev/null 2>&1; then
        rm -f "$tmpfile" "$tmpfile.c"
        return 0
    else
        rm -f "$tmpfile" "$tmpfile.c"
        return 1
    fi
}

# 检查库函数
check_function() {
    local func="$1"
    local tmpfile="/tmp/conftest.$$"
    
    cat > "$tmpfile.c" << EOF
void $func(void);
int main() { return 0; }
EOF
    
    if "$CC" $CFLAGS -o "$tmpfile" "$tmpfile.c" >/dev/null 2>&1; then
        rm -f "$tmpfile" "$tmpfile.c"
        return 0
    else
        rm -f "$tmpfile" "$tmpfile.c"
        return 1
    fi
}

# 检查库
check_library() {
    local lib="$1"
    local func="$2"
    local tmpfile="/tmp/conftest.$$"
    
    cat > "$tmpfile.c" << EOF
void $func(void);
int main() { $func(); return 0; }
EOF
    
    if "$CC" $CFLAGS -o "$tmpfile" "$tmpfile.c -l$lib" >/dev/null 2>&1; then
        LIBS="$LIBS -l$lib"
        rm -f "$tmpfile" "$tmpfile.c"
        return 0
    else
        rm -f "$tmpfile" "$tmpfile.c"
        return 1
    fi
}

# 特性检测
check_features() {
    log_info "检查系统特性"
    
    # 检查标准头文件
    for header in stdio.h stdlib.h string.h unistd.h; do
        if check_header "$header"; then
            echo "#define HAVE_$(echo $header | tr 'a-z.' 'A-Z_') 1" >> config.h.tmp
        fi
    done
    
    # 检查数学库
    if check_library m sqrt; then
        echo "#define HAVE_LIBM 1" >> config.h.tmp
    fi
    
    # 检查 readline
    if [ "$with_readline" != "no" ]; then
        if check_library readline readline; then
            echo "#define HAVE_READLINE 1" >> config.h.tmp
            log_info "找到 readline 库"
        elif [ "$with_readline" = "yes" ]; then
            log_error "readline 库未找到但被请求"
            exit 1
        else
            log_info "未找到 readline 库"
        fi
    fi
}

# 生成 config.h
generate_config_header() {
    cat > config.h << EOF
/* config.h - 由 configure 脚本生成 */
#ifndef CONFIG_H
#define CONFIG_H

/* 包信息 */
#define PACKAGE_NAME "$PACKAGE_NAME"
#define PACKAGE_VERSION "$PACKAGE_VERSION"
#define PACKAGE_BUGREPORT "$PACKAGE_BUGREPORT"

/* 安装目录 */
#define PREFIX "$prefix"
#define BINDIR "$bindir"

/* 编译器信息 */
#define CC "$CC"
#define CFLAGS "$CFLAGS"

/* 系统特性 */
EOF
    
    [ -f config.h.tmp ] && cat config.h.tmp >> config.h
    rm -f config.h.tmp
    
    cat >> config.h << EOF

#endif /* CONFIG_H */
EOF
}

# 生成 Makefile
generate_makefile() {
    cat > Makefile << EOF
# Makefile - 由 configure 脚本生成

# 编译器设置
CC = $CC
CFLAGS = $CFLAGS
LDFLAGS = $LDFLAGS
LIBS = $LIBS

# 安装目录
prefix = $prefix
exec_prefix = $exec_prefix
bindir = $bindir

# 调试设置
$(if [ "$enable_debug" = "yes" ]; then 
echo "DEBUG = -g -DDEBUG=1"
echo "CFLAGS += \$(DEBUG)"
fi)

# 源文件
SRCS = src/main.c src/util.c
OBJS = \$(SRCS:.c=.o)
TARGET = $PACKAGE_NAME

# 构建规则
all: \$(TARGET)

\$(TARGET): \$(OBJS)
	\$(CC) \$(LDFLAGS) -o \$@ \$(OBJS) \$(LIBS)

%.o: %.c
	\$(CC) \$(CFLAGS) -c -o \$@ $<

install: \$(TARGET)
	install -d \$(bindir)
	install \$(TARGET) \$(bindir)

clean:
	rm -f \$(OBJS) \$(TARGET)

distclean: clean
	rm -f config.h Makefile

.PHONY: all install clean distclean
EOF
}

# 主函数
main() {
    log_info "配置 $PACKAGE_NAME $PACKAGE_VERSION"
    
    parse_arguments "$@"
    check_system
    check_features
    generate_config_header
    generate_makefile
    
    # 显示配置摘要
    cat << EOF

配置摘要:
  名称:        $PACKAGE_NAME $PACKAGE_VERSION
  安装前缀:    $prefix
  编译器:      $CC
  编译器标志:  $CFLAGS
  调试模式:    $enable_debug
  Readline:    $with_readline

现在运行:
  make        # 编译程序
  make install # 安装程序
EOF
}

# 运行主函数
main "$@"

5. 使用方式

chmod +x configure

mkdir -p src

总结
• configure 是 Bash 脚本：用于系统检测和配置生成
• 两种编写方式：
  • 手动编写完整的 configure 脚本
  • 使用 Autotools（configure.ac + automake）
• 主要功能：参数解析、系统检测、文件生成
• 标准实践：对于复杂项目，推荐使用 Autotools 方式
对于简单的项目，手动编写configure脚本就足够了。对于需要广泛移植的复杂项目，建议使用标准的Autotools工具链。
