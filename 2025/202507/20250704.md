## ✅ 步骤一：自己源码安装 tinyrpc

假设你是通过以下方式安装的：

```bash
git clone https://github.com/xxx/tinyrpc.git
cd tinyrpc
mkdir build && cd build
cmake ..
make -j$(nproc)
sudo make install
```

默认情况下，`make install` 会把头文件放到 `/usr/local/include/tinyrpc`，库文件放到 `/usr/local/lib/libtinyrpc.a` 或 `.so/.dylib`。

---

## ✅ 步骤二：创建 `cmake/FindTinyRPC.cmake`

将以下内容保存为 `cmake/FindTinyRPC.cmake` 文件：

```cmake
# FindTinyRPC.cmake
# 查找 tinyrpc 库，并设置相关变量

# 查找头文件位置
find_path(TINYRPC_INCLUDE_DIR
    NAMES tinyrpc/common/log.h  # 根据你的实际头文件路径修改
    PATHS /usr/local/include
    DOC "tinyrpc 头文件目录"
)

# 查找库文件
find_library(TINYRPC_LIBRARY
    NAMES tinyrpc
    PATHS /usr/local/lib
    DOC "tinyrpc 库文件"
)

include(FindPackageHandleStandardArgs)

# 使用标准方式处理查找结果
find_package_handle_standard_args(TinyRPC
    REQUIRED_VARS TINYRPC_INCLUDE_DIR TINYRPC_LIBRARY
    VERSION_VAR TINYRPC_VERSION_STRING
)

if(TINYRPC_FOUND)
    message(STATUS "✅ 找到 tinyrpc")
    message(STATUS "    包含路径: ${TINYRPC_INCLUDE_DIR}")
    message(STATUS "    库路径: ${TINYRPC_LIBRARY}")
else()
    message(FATAL_ERROR "
❌ 错误：未找到 tinyrpc 库。
请确保你已经成功从源码安装 tinyrpc：
    git clone https://github.com/xxx/tinyrpc.git
    cd tinyrpc && mkdir build && cd build
    cmake .. && make && sudo make install
")
endif()
```

## ✅ 步骤三：在主 `CMakeLists.txt` 中使用该模块

```cmake
cmake_minimum_required(VERSION 3.14)
project(myproject)

# 添加 cmake 模块路径
set(CMAKE_MODULE_PATH "${PROJECT_SOURCE_DIR}/cmake" ${CMAKE_MODULE_PATH})

# 查找 tinyrpc
include(FindTinyRPC)

# 添加可执行文件
add_executable(${PROJECT_NAME} src/main.cpp)

# 链接 tinyrpc
target_include_directories(${PROJECT_NAME} PRIVATE ${TINYRPC_INCLUDE_DIR})
target_link_libraries(${PROJECT_NAME} PRIVATE ${TINYRPC_LIBRARY})
```

| 功能 | 实现方式 |
