ANTLR4符号表相关的API主要涉及符号解析、作用域管理和类型检查。以下是主要的API和概念：
1. 符号表基本接口

Symbol 接口

public interface Symbol {
    String getName();
    Scope getScope();
    Type getType();
}


Scope 接口

public interface Scope {
    String getScopeName();
    Scope getEnclosingScope();
    void define(Symbol sym);
    Symbol resolve(String name);
    Symbol resolveMember(String name);
}


2. 内置作用域实现

BaseScope - 基础作用域

public abstract class BaseScope implements Scope {
    protected Scope enclosingScope;
    protected Map<String, Symbol> symbols = new LinkedHashMap<>();
    
    public BaseScope(Scope enclosingScope) {
        this.enclosingScope = enclosingScope;
    }
    
    @Override
    public Symbol resolve(String name) {
        Symbol s = symbols.get(name);
        if (s != null) return s;
        if (enclosingScope != null) return enclosingScope.resolve(name);
        return null;
    }
    
    @Override
    public void define(Symbol sym) {
        symbols.put(sym.getName(), sym);
        sym.setScope(this);
    }
}


GlobalScope - 全局作用域

public class GlobalScope extends BaseScope {
    public GlobalScope() {
        super(null);
    }
    
    @Override
    public String getScopeName() {
        return "global";
    }
}


LocalScope - 局部作用域

public class LocalScope extends BaseScope {
    public LocalScope(Scope parent) {
        super(parent);
    }
    
    @Override
    public String getScopeName() {
        return "local";
    }
}


3. 符号类型

VariableSymbol - 变量符号

public class VariableSymbol extends BaseSymbol {
    public VariableSymbol(String name, Type type) {
        super(name, type);
    }
}


FunctionSymbol - 函数符号

public class FunctionSymbol extends BaseSymbol implements Scope {
    protected Map<String, Symbol> parameters = new LinkedHashMap<>();
    protected Scope enclosingScope;
    
    public FunctionSymbol(String name, Type returnType, Scope enclosingScope) {
        super(name, returnType);
        this.enclosingScope = enclosingScope;
    }
    
    public void defineParameter(Symbol param) {
        parameters.put(param.getName(), param);
    }
    
    @Override
    public Symbol resolve(String name) {
        Symbol s = parameters.get(name);
        if (s != null) return s;
        if (getEnclosingScope() != null) return getEnclosingScope().resolve(name);
        return null;
    }
}


4. 符号解析器

DefPhase - 定义收集阶段

public class DefPhase extends MyLanguageBaseListener {
    protected Scope currentScope;
    protected GlobalScope globals;
    
    @Override
    public void enterProgram(MyLanguageParser.ProgramContext ctx) {
        globals = new GlobalScope();
        currentScope = globals;
    }
    
    @Override
    public void enterFunctionDecl(MyLanguageParser.FunctionDeclContext ctx) {
        String name = ctx.ID().getText();
        FunctionSymbol function = new FunctionSymbol(name, null, currentScope);
        currentScope.define(function);
        currentScope = function; // 进入函数作用域
    }
    
    @Override
    public void exitFunctionDecl(MyLanguageParser.FunctionDeclContext ctx) {
        currentScope = currentScope.getEnclosingScope(); // 退出函数作用域
    }
    
    @Override
    public void enterVarDecl(MyLanguageParser.VarDeclContext ctx) {
        String name = ctx.ID().getText();
        VariableSymbol var = new VariableSymbol(name, null);
        currentScope.define(var);
    }
}


RefPhase - 引用解析阶段

public class RefPhase extends MyLanguageBaseListener {
    protected Scope currentScope;
    protected ParseTreeProperty<Scope> scopes;
    protected ParseTreeProperty<Symbol> symbols;
    
    @Override
    public void enterVarRef(MyLanguageParser.VarRefContext ctx) {
        String name = ctx.ID().getText();
        Symbol sym = currentScope.resolve(name);
        if (sym == null) {
            // 错误：未定义的变量
            System.err.println("Error: undefined variable " + name);
        }
        symbols.put(ctx, sym);
    }
}


5. 类型系统

Type - 类型接口

public interface Type {
    String getName();
    boolean isAssignableFrom(Type other);
}

public class BuiltInType implements Type {
    public static final BuiltInType INT = new BuiltInType("int");
    public static final BuiltInType FLOAT = new BuiltInType("float");
    public static final BuiltInType STRING = new BuiltInType("string");
    public static final BuiltInType BOOL = new BuiltInType("bool");
    public static final BuiltInType VOID = new BuiltInType("void");
    
    private final String name;
    
    public BuiltInType(String name) {
        this.name = name;
    }
    
    @Override
    public String getName() {
        return name;
    }
    
    @Override
    public boolean isAssignableFrom(Type other) {
        return this.equals(other);
    }
}


6. 完整的符号表构建示例

public class SymbolTableBuilder extends MyLanguageBaseListener {
    private Scope currentScope;
    private final ParseTreeProperty<Scope> scopes = new ParseTreeProperty<>();
    private final ParseTreeProperty<Type> types = new ParseTreeProperty<>();
    private final ParseTreeProperty<Symbol> symbols = new ParseTreeProperty<>();
    
    @Override
    public void enterProgram(MyLanguageParser.ProgramContext ctx) {
        GlobalScope global = new GlobalScope();
        scopes.put(ctx, global);
        currentScope = global;
    }
    
    @Override
    public void enterFunctionDecl(MyLanguageParser.FunctionDeclContext ctx) {
        String name = ctx.ID().getText();
        Type returnType = resolveType(ctx.type());
        
        FunctionSymbol function = new FunctionSymbol(name, returnType, currentScope);
        currentScope.define(function);
        
        scopes.put(ctx, function);
        currentScope = function;
        symbols.put(ctx, function);
    }
    
    @Override
    public void exitFunctionDecl(MyLanguageParser.FunctionDeclContext ctx) {
        currentScope = currentScope.getEnclosingScope();
    }
    
    @Override
    public void enterBlock(MyLanguageParser.BlockContext ctx) {
        currentScope = new LocalScope(currentScope);
        scopes.put(ctx, currentScope);
    }
    
    @Override
    public void exitBlock(MyLanguageParser.BlockContext ctx) {
        currentScope = currentScope.getEnclosingScope();
    }
    
    @Override
    public void enterVarDecl(MyLanguageParser.VarDeclContext ctx) {
        String name = ctx.ID().getText();
        Type type = resolveType(ctx.type());
        
        VariableSymbol var = new VariableSymbol(name, type);
        currentScope.define(var);
        symbols.put(ctx, var);
        types.put(ctx, type);
    }
    
    private Type resolveType(MyLanguageParser.TypeContext typeCtx) {
        if (typeCtx == null) return BuiltInType.VOID;
        String typeName = typeCtx.getText();
        switch (typeName) {
            case "int": return BuiltInType.INT;
            case "float": return BuiltInType.FLOAT;
            case "string": return BuiltInType.STRING;
            case "bool": return BuiltInType.BOOL;
            default: return BuiltInType.VOID;
        }
    }
    
    // 解析变量引用
    @Override
    public void enterVarRef(MyLanguageParser.VarRefContext ctx) {
        String name = ctx.ID().getText();
        Symbol sym = currentScope.resolve(name);
        if (sym != null) {
            symbols.put(ctx, sym);
            types.put(ctx, sym.getType());
        } else {
            System.err.println("Error: undefined symbol " + name);
        }
    }
}

MyLanguageLexer lexer = new MyLanguageLexer(input);

MyLanguageParser parser = new MyLanguageParser(tokens);

// 获取符号信息
