```
#pragma once
#include <string>
#include <optional>

class KVStore {
public:
    virtual ~KVStore() = default;
    
    // CRUD 操作
    virtual bool put(const std::string& key, 
                    const std::string& value) = 0;
    
    virtual std::optional<std::string> 
    get(const std::string& key) = 0;
    
    virtual bool del(const std::string& key) = 0;
    
    // 范围查询
    virtual std::vector<std::pair<std::string, std::string>>
    scan(const std::string& start, 
         const std::string& end) = 0;
};
```

2. 简单的内存存储 (src/storage/memory_store.h):
```
#pragma once
#include "kv_store.h"
#include <map>
#include <shared_mutex>

class MemoryStore : public KVStore {
    std::map<std::string, std::string> data_;
    mutable std::shared_mutex mutex_;
    
public:
    bool put(const std::string& key, 
             const std::string& value) override {
        std::unique_lock lock(mutex_);
        data_[key] = value;
        return true;
    }
    
    std::optional<std::string> 
    get(const std::string& key) override {
        std::shared_lock lock(mutex_);
        auto it = data_.find(key);
        if (it != data_.end()) {
            return it->second;
        }
        return std::nullopt;
    }
    
    bool del(const std::string& key) override {
        std::unique_lock lock(mutex_);
        return data_.erase(key) > 0;
    }
    
    std::vector<std::pair<std::string, std::string>>
    scan(const std::string& start, 
         const std::string& end) override {
        std::shared_lock lock(mutex_);
        std::vector<std::pair<std::string, std::string>> result;
        
        auto it_begin = data_.lower_bound(start);
        auto it_end = data_.upper_bound(end);
        
        for (auto it = it_begin; it != it_end; ++it) {
            result.emplace_back(it->first, it->second);
        }
        
        return result;
    }
};
```

3. 分片管理器 (src/sharding/shard_manager.h):
```
#pragma once
#include "storage/kv_store.h"
#include <vector>
#include <memory>
#include <functional>

class ShardManager {
    std::vector<std::unique_ptr<KVStore>> shards_;
    
    // 一致性哈希
    size_t getShardIndex(const std::string& key) {
        std::hash<std::string> hasher;
        return hasher(key) % shards_.size();
    }
    
public:
    ShardManager(size_t num_shards) {
        for (size_t i = 0; i < num_shards; ++i) {
            shards_.push_back(std::make_unique<MemoryStore>());
        }
    }
    
    bool put(const std::string& key, 
             const std::string& value) {
        size_t idx = getShardIndex(key);
        return shards_[idx]->put(key, value);
    }
    
    std::optional<std::string> 
    get(const std::string& key) {
        size_t idx = getShardIndex(key);
        return shards_[idx]->get(key);
    }
    
    bool del(const std::string& key) {
        size_t idx = getShardIndex(key);
        return shards_[idx]->del(key);
    }
    
    // 分布式扫描（跨分片）
    std::vector<std::pair<std::string, std::string>>
    distributedScan(const std::string& start, 
                    const std::string& end) {
        std::vector<std::pair<std::string, std::string>> result;
        
        // 并行查询所有分片
        for (auto& shard : shards_) {
            auto shard_result = shard->scan(start, end);
            result.insert(result.end(), 
                         shard_result.begin(), 
                         shard_result.end());
        }
        
        // 排序结果
        std::sort(result.begin(), result.end(),
                 [](const auto& a, const auto& b) {
                     return a.first < b.first;
                 });
        
        return result;
    }
};
```

4. 简单的 RPC 服务器 (src/network/rpc_server.h):
```
#pragma once
#include <functional>
#include <string>
#include <map>

class RPCServer {
    using Handler = std::function<std::string(const std::string&)>;
    std::map<std::string, Handler> handlers_;
    
public:
    void registerHandler(const std::string& method, 
                         Handler handler) {
        handlers_[method] = handler;
    }
    
    std::string handleRequest(const std::string& method,
                             const std::string& request) {
        auto it = handlers_.find(method);
        if (it != handlers_.end()) {
            return it->second(request);
        }
        return "ERROR: Method not found";
    }
};
```
