根本原因

time_of_day 不是 C++20 标准的一部分，它是一个GCC libstdc++内部的实现细节，不可移植。

    在某些 GCC 版本中（如 13.1），std::chrono 的内部实现中存在 time_of_day 这个别名，用于构造 hh_mm_ss。
    但在 Clang（使用 libc++）或较新/较旧的 GCC 中，它不存在。
    因此，当您在 GitHub Actions 上使用 Clang 时，time_of_day 未定义，编译失败。

## 步骤4：GitHub Actions 自动下载使用

在 `.github/workflows/your-workflow.yml` 中添加：

```yaml
name: Build with JDK 7u80

on: [push, pull_request]

jobs:
  build:
    runs-on: ubuntu-latest
    
    steps:
      - name: Checkout code
        uses: actions/checkout@v4
      
      - name: Download JDK 7u80 from Release
        id: download-jdk
        uses: actions/github-script@v7
        with:
          script: |
            // 获取指定 Release 的 assets
            const release = await github.rest.repos.getReleaseByTag({
              owner: context.repo.owner,
              repo: context.repo.repo,
              tag_name: 'jdk-7u80-v1.0'
            });
            
            // 找到 JDK 文件
            const jdkAsset = release.data.assets.find(asset => 
              asset.name === 'jdk-7u80-linux-x64.tar.gz'
            );
            
            if (!jdkAsset) {
              core.setFailed('JDK asset not found in release');
              return;
            }
            
            // 设置输出变量（下载链接）
            core.setOutput('download_url', jdkAsset.browser_download_url);
            core.setOutput('asset_id', jdkAsset.id.toString());
            core.info(`Found JDK: ${jdkAsset.name} (${jdkAsset.size} bytes)`);
      
      - name: Download and setup JDK
        run: |
          echo "Downloading JDK from: ${{ steps.download-jdk.outputs.download_url }}"
          
          # 下载 JDK（142MB）
          curl -L -o jdk-7u80-linux-x64.tar.gz \
               -H "Accept: application/octet-stream" \
               "${{ steps.download-jdk.outputs.download_url }}"
          
          # 验证文件大小
          ls -lh jdk-7u80-linux-x64.tar.gz
          test -s jdk-7u80-linux-x64.tar.gz || exit 1
          
          # 解压到 ~/jdk
          mkdir -p ~/jdk
          tar -xzf jdk-7u80-linux-x64.tar.gz -C ~/jdk --strip-components=1
          
          # 设置环境变量
          echo "JAVA_HOME=~/jdk" >> $GITHUB_ENV
          echo "$JAVA_HOME/bin" >> $GITHUB_PATH
          
          # 清理
          rm jdk-7u80-linux-x64.tar.gz
      
      - name: Verify JDK Installation
        run: |
          java -version
          javac -version
          echo "JDK setup completed successfully!"
      
      - name: Run your build
        run: |
          # 你的 Maven/Gradle/Ant 构建命令
          mvn clean compile -B  # 示例
```
